import tomllib


def read_config(path) -> tuple[int, int]:
    """Read integer timeout and retries values from a TOML file."""
    with open(path, "rb") as file:
        config = tomllib.load(file)

    timeout = config["timeout"]
    retries = config["retries"]

    for name, value in (("timeout", timeout), ("retries", retries)):
        if type(value) is not int:
            raise TypeError(f"{name} must be an integer")

    return timeout, retries

import tomllib

import pytest

from config import read_config


def test_read_config(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 30\nretries = 0\n", encoding="utf-8")

    assert read_config(path) == (30, 0)


@pytest.mark.parametrize("missing", ["timeout", "retries"])
def test_missing_key(tmp_path, missing):
    path = tmp_path / "config.toml"
    remaining = "retries" if missing == "timeout" else "timeout"
    path.write_text(f"{remaining} = 3\n", encoding="utf-8")

    with pytest.raises(KeyError) as exc:
        read_config(path)

    assert exc.value.args == (missing,)


@pytest.mark.parametrize("name", ["timeout", "retries"])
@pytest.mark.parametrize("value", ['"3"', "3.0", "true"])
def test_non_integer_value(tmp_path, name, value):
    path = tmp_path / "config.toml"
    values = {"timeout": "30", "retries": "3"}
    values[name] = value
    path.write_text(
        "\n".join(f"{key} = {item}" for key, item in values.items()),
        encoding="utf-8",
    )

    with pytest.raises(TypeError, match=f"^{name} must be an integer$"):
        read_config(path)


def test_invalid_toml(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = [", encoding="utf-8")

    with pytest.raises(tomllib.TOMLDecodeError):
        read_config(path)
