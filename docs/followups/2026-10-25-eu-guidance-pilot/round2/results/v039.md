import tomllib


def read_config(path):
    """Read a TOML file and return (timeout, retries) as a tuple of two ints."""
    with open(path, "rb") as f:
        data = tomllib.load(f)
    return int(data["timeout"]), int(data["retries"])

import pytest

from config import read_config


def test_read_config(tmp_path):
    p = tmp_path / "c.toml"
    p.write_text("timeout = 30\nretries = 3\n")
    assert read_config(p) == (30, 3)


def test_missing_key(tmp_path):
    p = tmp_path / "c.toml"
    p.write_text("timeout = 30\n")
    with pytest.raises(KeyError):
