import json
import sys


def decide(payload: str) -> str:
    """Determine whether to allow or deny a tool call payload.

    Returns 'deny' if the command contains the text 'rm -rf /'.
    Otherwise returns 'allow'.
    """
    try:
        data = json.loads(payload)
        if isinstance(data, dict):
            command = data.get("command")
            if isinstance(command, str) and "rm -rf /" in command:
                return "deny"
            return "allow"
    except Exception:
        if "rm -rf /" in payload:
            return "deny"
        return "allow"
    return "allow"


def main() -> None:
    if hasattr(sys.stdin, "buffer"):
        payload = sys.stdin.buffer.read().decode("utf-8", errors="replace")
    else:
        payload = sys.stdin.read()

    decision = decide(payload)
    print(json.dumps({"decision": decision}))


if __name__ == "__main__":
    main()

import json
import subprocess
import sys
from pathlib import Path

import hook

HOOK_PATH = Path(__file__).parent / "hook.py"


def run_hook_cli(input_data: str | bytes) -> subprocess.CompletedProcess[str]:
    """Run hook.py as a subprocess with the given input to stdin."""
    if isinstance(input_data, str):
        stdin_bytes = input_data.encode("utf-8")
    else:
        stdin_bytes = input_data

    return subprocess.run(
        [sys.executable, str(HOOK_PATH)],
        input=stdin_bytes,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        check=False,
    )


def test_allow_safe_command() -> None:
