import sys
from pathlib import Path


def count_lines(path: Path) -> int:
    with path.open("rb") as f:
        return sum(1 for _ in f)


def main(argv: list[str] | None = None) -> int:
    args = sys.argv[1:] if argv is None else argv
    for name in args:
        count = count_lines(Path(name))
        print(f"{count} {name}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from pathlib import Path
import subprocess
import sys

import pytest

from count_lines import count_lines, main


def test_count_lines_multiple_files(tmp_path: Path, capsys: pytest.CaptureFixture[str]) -> None:
    empty = tmp_path / "empty.txt"
    empty.write_text("", encoding="utf-8")

    three = tmp_path / "three.txt"
    three.write_text("a\nb\nc\n", encoding="utf-8")

    rc = main([str(empty), str(three)])
    captured = capsys.readouterr()

    assert rc == 0
    assert captured.out == f"0 {empty}\n3 {three}\n"


def test_cli_invocation(tmp_path: Path) -> None:
    sample = tmp_path / "sample.txt"
    sample.write_text("first\nsecond\n", encoding="utf-8")

    result = subprocess.run(
        [sys.executable, "count_lines.py", str(sample)],
        capture_output=True,
        text=True,
        check=True,
    )
    assert result.stdout == f"2 {sample}\n"


def test_count_lines_without_trailing_newline(tmp_path: Path) -> None:
    no_nl = tmp_path / "no_nl.txt"
    no_nl.write_text("line1\nline2", encoding="utf-8")

    assert count_lines(no_nl) == 2
