```python
import tomllib


def read_config(path):
    with open(path, "rb") as f:
        data = tomllib.load(f)
    timeout = data["timeout"]
    retries = data["retries"]
    if type(timeout) is not int or type(retries) is not int:
        raise ValueError("timeout and retries must be integers")
    return timeout, retries
```

```python
import pytest

from config import read_config


def test_returns_timeout_and_retries_as_ints(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 30\nretries = 3\n")
    result = read_config(path)
    assert result == (30, 3)
    assert isinstance(result, tuple)
    assert all(type(value) is int for value in result)


def test_ignores_other_keys_and_tables(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text('name = "service"\ntimeout = 5\nretries = 0\n\n[extra]\ntimeout = 99\n')
    assert read_config(path) == (5, 0)


def test_accepts_string_path(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 10\nretries = 2\n")
    assert read_config(str(path)) == (10, 2)


def test_missing_key_raises_key_error(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 10\n")
    with pytest.raises(KeyError):
        read_config(path)


@pytest.mark.parametrize("body", [
    'timeout = "30"\nretries = 3\n',
    "timeout = 2.5\nretries = 3\n",
    "timeout = 30\nretries = true\n",
])
def test_non_integer_value_raises_value_error(tmp_path, body):
    path = tmp_path / "config.toml"
    path.write_text(body)
    with pytest.raises(ValueError):
        read_config(path)


def test_missing_file_raises_file_not_found(tmp_path):
    with pytest.raises(FileNotFoundError):
```
