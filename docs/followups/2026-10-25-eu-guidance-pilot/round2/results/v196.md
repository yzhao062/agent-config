"""Print the number of lines in each file named on the command line."""

import sys


def main(argv):
    for name in argv:
        with open(name, "rb") as handle:
            count = sum(1 for _ in handle)
        print(f"{count} {name}")


if __name__ == "__main__":
    main(sys.argv[1:])

import subprocess
import sys
from pathlib import Path

SCRIPT = Path(__file__).parent / "count_lines.py"


def run(*names):
    result = subprocess.run(
        [sys.executable, str(SCRIPT), *map(str, names)],
        capture_output=True,
        text=True,
        check=True,
    )
    return result.stdout.splitlines()


def test_single_file(tmp_path):
    path = tmp_path / "a.txt"
    path.write_text("one\ntwo\nthree\n")
    assert run(path) == [f"3 {path}"]


def test_multiple_files_in_order(tmp_path):
    first = tmp_path / "a.txt"
    second = tmp_path / "b.txt"
    first.write_text("x\n")
    second.write_text("x\ny\n")
    assert run(first, second) == [f"1 {first}", f"2 {second}"]


def test_empty_file(tmp_path):
    path = tmp_path / "empty.txt"
    path.write_text("")
    assert run(path) == [f"0 {path}"]


def test_last_line_without_newline(tmp_path):
    path = tmp_path / "a.txt"
    path.write_text("one\ntwo")
    assert run(path) == [f"2 {path}"]
