#!/usr/bin/env python3
"""Count lines in files specified on the command line."""

import os
import sys


def count_lines(file_path: str | os.PathLike[str]) -> int:
    """Return the number of lines in the file at file_path."""
    with open(file_path, "r", encoding="utf-8", errors="replace") as file:
        return sum(1 for _ in file)


def main(argv: list[str] | None = None) -> int:
    """Print the line count for each file provided in arguments."""
    if argv is None:
        argv = sys.argv[1:]

    exit_code = 0
    for file_path in argv:
        try:
            line_count = count_lines(file_path)
            print(f"{line_count} {file_path}")
        except OSError as error:
            error_message = error.strerror if error.strerror else str(error)
            print(f"count_lines.py: {file_path}: {error_message}", file=sys.stderr)
            exit_code = 1

    return exit_code


if __name__ == "__main__":
    sys.exit(main())
