```python
def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge two dictionaries and return a new dictionary."""
    merged = dict(base)
    for key, value in override.items():
        if key in merged and isinstance(merged[key], dict) and isinstance(value, dict):
            merged[key] = deep_merge(merged[key], value)
        else:
            merged[key] = value
    return merged
```

```python
from merge import deep_merge


def test_flat_merge_and_override():
    base = {"host": "localhost", "port": 8080}
    override = {"port": 9090, "debug": True}

    result = deep_merge(base, override)

    assert result == {"host": "localhost", "port": 9090, "debug": True}


def test_recursive_dict_merge():
    base = {
        "database": {"host": "localhost", "port": 5432, "options": {"timeout": 30}},
        "logging": {"level": "INFO"},
    }
    override = {
        "database": {"port": 5433, "options": {"retries": 3}},
    }

    result = deep_merge(base, override)

    assert result == {
        "database": {
            "host": "localhost",
            "port": 5433,
            "options": {"timeout": 30, "retries": 3},
        },
        "logging": {"level": "INFO"},
    }


def test_override_replaces_mismatched_types():
    base = {"section": {"nested": 1}, "scalar": "text", "items": [1, 2]}
    override = {"section": "disabled", "scalar": {"nested": 2}, "items": [3]}

    result = deep_merge(base, override)

    assert result == {
        "section": "disabled",
        "scalar": {"nested": 2},
        "items": [3],
    }


def test_inputs_are_not_mutated():
    base = {"nested": {"a": 1, "b": 2}, "top": "initial"}
    override = {"nested": {"b": 3, "c": 4}, "extra": True}

    _ = deep_merge(base, override)

    assert base == {"nested": {"a": 1, "b": 2}, "top": "initial"}
    assert override == {"nested": {"b": 3, "c": 4}, "extra": True}
```
