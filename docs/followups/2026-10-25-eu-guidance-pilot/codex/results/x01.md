import json
import sys


def main() -> int:
    try:
        payload = json.load(sys.stdin)
    except (json.JSONDecodeError, UnicodeDecodeError):
        payload = {}

    command = payload.get("command", "") if isinstance(payload, dict) else ""
    denied = isinstance(command, str) and "rm -rf /" in command
    print(json.dumps({"decision": "deny" if denied else "allow"}))
    return 0


if __name__ == "__main__":
    sys.exit(main())

import json
from pathlib import Path
import subprocess
import sys

import pytest


HOOK = Path(__file__).with_name("hook.py")


@pytest.mark.parametrize(
    ("payload", "decision"),
    [
        ('{"tool": "Bash", "command": "ls -la"}', "allow"),
        ('{"tool": "Bash", "command": "rm -rf /"}', "deny"),
        ('{"command": "echo hello; rm -rf /tmp/example"}', "deny"),
        ('{"command": "echo \\"rm -rf /\\""}', "deny"),
        ('{"command": "rm -rf relative/path"}', "allow"),
        ('{"command": "RM -RF /"}', "allow"),
        ("", "allow"),
        ("   \n", "allow"),
        ('{"tool": "Bash", "command":', "allow"),
        ("invalid JSON", "allow"),
        ("{}", "allow"),
        ("null", "allow"),
        ("[]", "allow"),
        ('{"command": null}', "allow"),
        ('{"command": 42}', "allow"),
    ],
)
def test_hook(payload: str, decision: str) -> None:
    result = subprocess.run(
        [sys.executable, str(HOOK)],
        input=payload,
        text=True,
        capture_output=True,
        timeout=5,
        check=False,
    )

    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": decision}
    assert result.stderr == ""
