```python
import tomllib
from os import PathLike


def read_config(path: str | PathLike[str]) -> tuple[int, int]:
    with open(path, "rb") as file:
        config = tomllib.load(file)

    timeout = config["timeout"]
    retries = config["retries"]

    if type(timeout) is not int or type(retries) is not int:
        raise TypeError("timeout and retries must both be integers")

    return timeout, retries
```

```python
import tomllib

import pytest

from config import read_config


@pytest.mark.parametrize("use_string_path", [False, True])
def test_read_config(tmp_path, use_string_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")

    result = read_config(str(path) if use_string_path else path)

    assert result == (30, 3)
    assert isinstance(result, tuple)
    assert all(type(value) is int for value in result)


@pytest.mark.parametrize("key", ["timeout", "retries"])
@pytest.mark.parametrize("value", ['"3"', "3.0", "true", "[3]"])
def test_rejects_non_integer_values(tmp_path, key, value):
    values = {"timeout": "30", "retries": "3"}
    values[key] = value
    path = tmp_path / "config.toml"
    path.write_text(
        "\n".join(f"{name} = {item}" for name, item in values.items()),
        encoding="utf-8",
    )

    with pytest.raises(TypeError, match="must both be integers"):
        read_config(path)


@pytest.mark.parametrize("missing_key", ["timeout", "retries"])
def test_missing_key(tmp_path, missing_key):
    values = {"timeout": 30, "retries": 3}
    del values[missing_key]
    path = tmp_path / "config.toml"
    path.write_text(
        "\n".join(f"{name} = {value}" for name, value in values.items()),
        encoding="utf-8",
    )

    with pytest.raises(KeyError) as error:
        read_config(path)

    assert error.value.args == (missing_key,)


def test_invalid_toml(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = [\n", encoding="utf-8")

    with pytest.raises(tomllib.TOMLDecodeError):
        read_config(path)


def test_missing_file(tmp_path):
    with pytest.raises(FileNotFoundError):
        read_config(tmp_path / "missing.toml")
```
