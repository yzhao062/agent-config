import json
import sys


def main() -> None:
    raw = sys.stdin.read().strip()
    command = ""
    if raw:
        try:
            payload = json.loads(raw)
        except json.JSONDecodeError:
            payload = {}
        if isinstance(payload, dict) and isinstance(payload.get("command"), str):
            command = payload["command"]

    decision = "deny" if "rm -rf /" in command else "allow"
    print(json.dumps({"decision": decision}))


if __name__ == "__main__":
    main()

import json
from pathlib import Path
import subprocess
import sys

HOOK_PATH = Path(__file__).with_name("hook.py")


def test_allows_safe_command() -> None:
    result = subprocess.run(
        [sys.executable, str(HOOK_PATH)],
        input='{"tool": "Bash", "command": "ls -la"}',
        text=True,
        capture_output=True,
        check=False,
    )
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "allow"}
