from pathlib import Path
import sys


def main(argv: list[str]) -> int:
    if not argv:
        sys.stderr.write("Usage: count_lines.py <file> [<file> ...]\n")
        return 1

    for name in argv:
        path = Path(name)
        if not path.is_file():
            sys.stderr.write(f"Error: {name} is not a readable file\n")
            return 1

    for name in argv:
        with open(name, "r", encoding="utf-8") as handle:
            count = sum(1 for _ in handle)
        print(f"{count} {name}")

    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))

from pathlib import Path
import subprocess
import sys

from count_lines import main


SCRIPT_PATH = Path(__file__).with_name("count_lines.py")


def test_single_file(tmp_path: Path, capsys) -> None:
    target = tmp_path / "sample.txt"
    target.write_text("alpha\nbeta\ngamma\n", encoding="utf-8")

    exit_code = main([str(target)])
    captured = capsys.readouterr()

    assert exit_code == 0
    assert captured.out == f"3 {target}\n"
    assert captured.err == ""


def test_multiple_files_and_empty_file(tmp_path: Path) -> None:
    first = tmp_path / "first.txt"
    second = tmp_path / "empty.txt"
    first.write_text("one\ntwo\n", encoding="utf-8")
    second.write_text("", encoding="utf-8")

    result = subprocess.run(
        [sys.executable, str(SCRIPT_PATH), str(first), str(second)],
        capture_output=True,
        text=True,
        check=False,
    )

    assert result.returncode == 0
    assert result.stdout == f"2 {first}\n0 {second}\n"
    assert result.stderr == ""


def test_missing_file_rejected_at_boundary(tmp_path: Path, capsys) -> None:
    missing = tmp_path / "missing.txt"

    exit_code = main([str(missing)])
    captured = capsys.readouterr()

    assert exit_code == 1
    assert captured.out == ""
    assert f"Error: {missing} is not a readable file" in captured.err
