#!/usr/bin/env python3
"""A command-line script to count lines in files."""

from __future__ import annotations

import argparse
import os
import sys


def count_lines(filepath: str | os.PathLike[str]) -> int:
    """Count and return the number of lines in a file.

    An empty file contains 0 lines. Files with or without a trailing
    newline are counted by the number of line entries yielded.
    """
    with open(filepath, "r", encoding="utf-8", errors="replace") as file_handle:
        return sum(1 for _ in file_handle)


def main(argv: list[str] | None = None) -> int:
    """Execute the command-line interface for counting file lines."""
    parser = argparse.ArgumentParser(
        description="Print the number of lines in each file named on the command line."
    )
    parser.add_argument(
        "files",
        nargs="*",
        metavar="FILE",
        help="Path to one or more files to count.",
    )
    arguments = parser.parse_args(argv)

    exit_status = 0
    for filepath in arguments.files:
        try:
            line_count = count_lines(filepath)
            print(f"{line_count} {filepath}")
        except OSError as error:
            print(f"count_lines: {filepath}: {error}", file=sys.stderr)
            exit_status = 1

    return exit_status


if __name__ == "__main__":
    sys.exit(main())
