"""Print the number of lines in each file named on the command line."""

import sys


def count_lines(path):
    """Return the number of lines in the file at path."""
    with open(path, "rb") as handle:
        return sum(1 for _ in handle)


def main(argv=None):
    """Print one '<count> <name>' line per file. Return an exit status."""
    args = sys.argv[1:] if argv is None else argv
    status = 0
    for name in args:
        try:
            print(f"{count_lines(name)} {name}")
        except OSError as error:
            print(f"count_lines: cannot read {name}: {error.strerror}", file=sys.stderr)
            status = 1
    return status


if __name__ == "__main__":
    sys.exit(main())

import count_lines


def test_counts_lines(tmp_path, capsys):
    first = tmp_path / "a.txt"
    second = tmp_path / "b.txt"
    first.write_text("one\ntwo\nthree\n")
    second.write_text("only")
    assert count_lines.main([str(first), str(second)]) == 0
    out = capsys.readouterr().out.splitlines()
    assert out == [f"3 {first}", f"1 {second}"]


def test_empty_file(tmp_path, capsys):
    path = tmp_path / "empty.txt"
    path.write_text("")
    assert count_lines.main([str(path)]) == 0
    assert capsys.readouterr().out == f"0 {path}\n"


def test_missing_file_reports_error(tmp_path, capsys):
    good = tmp_path / "g.txt"
    good.write_text("x\n")
    missing = tmp_path / "missing.txt"
    assert count_lines.main([str(missing), str(good)]) == 1
    captured = capsys.readouterr()
    assert captured.out == f"1 {good}\n"
    assert "missing.txt" in captured.err
