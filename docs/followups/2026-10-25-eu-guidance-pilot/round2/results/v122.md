```python
def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge two dictionaries without mutating either input."""
    merged = dict(base)
    for key, value in override.items():
        if key in merged and isinstance(merged[key], dict) and isinstance(value, dict):
            merged[key] = deep_merge(merged[key], value)
        else:
            merged[key] = value
    return merged
```

```python
from merge import deep_merge


def test_disjoint_and_overridden_keys():
    base = {"host": "localhost", "port": 8080}
    override = {"port": 9090, "debug": True}

    result = deep_merge(base, override)

    assert result == {"host": "localhost", "port": 9090, "debug": True}


def test_recursive_dict_merge():
    base = {
        "database": {"host": "localhost", "port": 5432, "options": {"timeout": 10}},
        "logging": {"level": "INFO"},
    }
    override = {
        "database": {"port": 5433, "options": {"retries": 3}},
    }

    result = deep_merge(base, override)

    assert result == {
        "database": {
            "host": "localhost",
            "port": 5433,
            "options": {"timeout": 10, "retries": 3},
        },
        "logging": {"level": "INFO"},
    }


def test_type_mismatch_override_wins():

        "section_to_scalar": False,
        "scalar_to_section": {"mode": "custom"},
        "list_value": [3],

    result = deep_merge(base, override)

    assert result == {

        "scalar_to_section": {"mode": "custom"},

    }


def test_inputs_are_not_mutated():
    base = {"nested": {"a": 1, "b": 2}, "top": "keep"}
    override = {"nested": {"b": 3, "c": 4}, "extra": "new"}

    result = deep_merge(base, override)

    assert base == {"nested": {"a": 1, "b": 2}, "top": "keep"}
    assert override == {"nested": {"b": 3, "c": 4}, "extra": "new"}
    assert result is not base
    assert result is not override
    assert result["nested"] is not base["nested"]
```
