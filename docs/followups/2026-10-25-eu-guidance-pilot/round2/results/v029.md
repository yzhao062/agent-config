```python
"""Pre-tool-call hook: deny commands containing 'rm -rf /', allow everything else.

The hook always prints a decision and exits with code 0, even when the payload
is empty, truncated, or otherwise malformed, so the host never blocks a tool
call because of a hook failure.
"""
import json
import sys

DANGEROUS = "rm -rf /"


def decide(raw: str) -> str:
    try:
        payload = json.loads(raw)
    except (ValueError, TypeError):
        # Empty or truncated payload: fall back to scanning the raw text.
        return "deny" if DANGEROUS in (raw or "") else "allow"
    if isinstance(payload, dict):
        command = payload.get("command", "")
        text = command if isinstance(command, str) else json.dumps(command)
    else:
        text = raw
    return "deny" if DANGEROUS in text else "allow"


def main() -> int:
    try:
        raw = sys.stdin.read()
    except Exception:
        raw = ""
    try:
        decision = decide(raw)
    except Exception:
        decision = "allow"
    print(json.dumps({"decision": decision}))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
import json
import subprocess
import sys
from pathlib import Path

import pytest

HOOK = str(Path(__file__).parent / "hook.py")


def run(stdin: str):
    proc = subprocess.run([sys.executable, HOOK], input=stdin,
                          capture_output=True, text=True)
    return proc.returncode, proc.stdout


@pytest.mark.parametrize("payload,expected", [
    ({"tool": "Bash", "command": "ls -la"}, "allow"),
    ({"tool": "Bash", "command": "rm -rf /"}, "deny"),
    ({"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"}, "deny"),
    ({"tool": "Bash", "command": "rm -rf ./build"}, "allow"),
    ({"tool": "Read"}, "allow"),
    ({"tool": "Bash", "command": None}, "allow"),
])
def test_json_payloads(payload, expected):
    code, out = run(json.dumps(payload))
    assert code == 0
    assert json.loads(out) == {"decision": expected}


@pytest.mark.parametrize("raw", ["", "   \n", "{", '{"tool": "Bash", "comm', "null", "[]", "not json"])
def test_bad_payloads_allow_and_exit_zero(raw):
    code, out = run(raw)
    assert code == 0
    assert json.loads(out) == {"decision": "allow"}


def test_truncated_payload_with_dangerous_text_denied():
    code, out = run('{"tool": "Bash", "command": "rm -rf /')
    assert code == 0
    assert json.loads(out) == {"decision": "deny"}
```
