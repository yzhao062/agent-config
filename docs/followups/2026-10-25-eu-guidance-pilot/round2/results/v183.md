"""Command-line script to count lines in files."""

from pathlib import Path
import sys
from typing import Sequence


def count_file_lines(path: str | Path) -> int:
    """Return the number of lines in the file at path."""
    with Path(path).open("r", encoding="utf-8") as file_handle:
        return sum(1 for _ in file_handle)


def main(argv: Sequence[str] | None = None) -> int:
    """Print '<count> <name>' for each file named in argv."""
    args = sys.argv[1:] if argv is None else list(argv)
    for filename in args:
        line_count = count_file_lines(filename)
        print(f"{line_count} {filename}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

"""Tests for count_lines.py."""

from pathlib import Path
import subprocess
import sys

import pytest

import count_lines


def test_count_file_lines_empty(tmp_path: Path) -> None:
    """An empty file has zero lines."""
    empty_file = tmp_path / "empty.txt"
    empty_file.write_text("", encoding="utf-8")
    assert count_lines.count_file_lines(empty_file) == 0


def test_count_file_lines_multiple(tmp_path: Path) -> None:
    """A multi-line file returns the total number of lines."""
    sample_file = tmp_path / "sample.txt"
    sample_file.write_text("first\nsecond\nthird\n", encoding="utf-8")
    assert count_lines.count_file_lines(sample_file) == 3


def test_main_prints_counts_in_order(
    tmp_path: Path, capsys: pytest.CaptureFixture[str]
) -> None:
    """main prints '<count> <name>' for each file in argument order."""
    first = tmp_path / "first.txt"
    second = tmp_path / "second.txt"
    first.write_text("a\nb\n", encoding="utf-8")
    second.write_text("only one line", encoding="utf-8")

    exit_code = count_lines.main([str(first), str(second)])
    captured = capsys.readouterr()

    assert exit_code == 0
    assert captured.out.splitlines() == [
        f"2 {first}",
        f"1 {second}",
    ]


def test_cli_execution(tmp_path: Path) -> None:
    """Running count_lines.py as a script outputs line counts."""
    target = tmp_path / "data.txt"
    target.write_text("line 1\nline 2\nline 3\nline 4\n", encoding="utf-8")
    script_path = Path(count_lines.__file__).resolve()

    result = subprocess.run(
        [sys.executable, str(script_path), str(target)],
