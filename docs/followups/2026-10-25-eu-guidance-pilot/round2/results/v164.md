"""Print the number of lines in each file named on the command line."""

import sys


def count_lines(path):
    with open(path, "rb") as handle:
        return sum(1 for _ in handle)


def main(argv):
    for name in argv:
        print(f"{count_lines(name)} {name}")


if __name__ == "__main__":
    main(sys.argv[1:])

import subprocess
import sys
from pathlib import Path

SCRIPT = Path(__file__).parent / "count_lines.py"


def run(*args):
    return subprocess.run(
        [sys.executable, str(SCRIPT), *map(str, args)],
        capture_output=True,
        text=True,
        check=True,
    ).stdout


def test_single_file(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"one\ntwo\nthree\n")
    assert run(f) == f"3 {f}\n"


def test_multiple_files_in_order(tmp_path):
    a = tmp_path / "a.txt"
    b = tmp_path / "b.txt"
    a.write_bytes(b"x\n")
    b.write_bytes(b"x\ny\n")
    assert run(a, b) == f"1 {a}\n2 {b}\n"


def test_empty_file(tmp_path):
    f = tmp_path / "empty.txt"
    f.write_bytes(b"")
    assert run(f) == f"0 {f}\n"


def test_last_line_without_newline(tmp_path):
    f = tmp_path / "n.txt"
    f.write_bytes(b"one\ntwo")
    assert run(f) == f"2 {f}\n"
