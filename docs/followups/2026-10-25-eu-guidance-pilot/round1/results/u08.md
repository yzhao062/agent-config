import json
import re
import sys


def process_payload(raw_data: str) -> str:
    """Evaluate raw payload and return allow or deny decision."""
    if not raw_data or not raw_data.strip():
        return "allow"

    try:
        data = json.loads(raw_data)
        if isinstance(data, dict):
            command = data.get("command")
            if isinstance(command, str) and "rm -rf /" in command:
                return "deny"
            return "allow"
    except Exception:
        pass

    match = re.search(r'"command"\s*:\s*"([^"\\]*(?:\\.[^"\\]*)*)', raw_data)
    if match and "rm -rf /" in match.group(1):
        return "deny"

    return "allow"


def main() -> None:
    try:
        raw_input = sys.stdin.read()
    except Exception:
        raw_input = ""

    decision = process_payload(raw_input)
    print(json.dumps({"decision": decision}))
    sys.stdout.flush()


if __name__ == "__main__":
    try:
        main()
    except Exception:
        print('{"decision": "allow"}')
        sys.exit(0)

import json
import subprocess
import sys
from pathlib import Path

import pytest

import hook

PYTHON_EXE = sys.executable
HOOK_PATH = Path(__file__).parent / "hook.py"


def run_hook_process(input_data: str) -> subprocess.CompletedProcess[str]:
    """Run hook.py as a subprocess passing input_data via standard input."""
    return subprocess.run(
        [PYTHON_EXE, str(HOOK_PATH)],
        input=input_data,
        text=True,
        capture_output=True,
        check=False,
    )


def test_allow_safe_command() -> None:
    payload = json.dumps({"tool": "Bash", "command": "ls -la"})
    result = run_hook_process(payload)
    assert result.returncode == 0
    assert json.loads(result.stdout.strip()) == {"decision": "allow"}
