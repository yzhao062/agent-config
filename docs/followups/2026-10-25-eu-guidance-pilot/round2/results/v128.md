import tomllib
from pathlib import Path


def read_config(path: str | Path) -> tuple[int, int]:
    """Read a TOML configuration file and return timeout and retries as a tuple of two ints."""
    with open(path, "rb") as file:
        data = tomllib.load(file)
    return int(data["timeout"]), int(data["retries"])

from pathlib import Path
import pytest
import tomllib
from config import read_config


def test_read_config_valid(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")

    result = read_config(config_file)

    assert result == (30, 3)
    assert isinstance(result[0], int)
    assert isinstance(result[1], int)


def test_read_config_str_path(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 10\nretries = 1\n", encoding="utf-8")

    result = read_config(str(config_file))

    assert result == (10, 1)
