```python
from typing import Any


def deep_merge(base: dict[str, Any], override: dict[str, Any]) -> dict[str, Any]:
    """Recursively merge two dictionaries without mutating either input."""
    merged: dict[str, Any] = dict(base)
    for key, override_value in override.items():
        base_value = merged.get(key)
        if isinstance(base_value, dict) and isinstance(override_value, dict):
            merged[key] = deep_merge(base_value, override_value)
        else:
            merged[key] = override_value
    return merged
```

```python
from copy import deepcopy

from merge import deep_merge


def test_disjoint_keys() -> None:
    base = {"a": 1}
    override = {"b": 2}
    assert deep_merge(base, override) == {"a": 1, "b": 2}


def test_override_scalar_and_list_values() -> None:
    base = {"a": 1, "items": [1, 2], "keep": "yes"}
    override = {"a": 2, "items": [3]}
    assert deep_merge(base, override) == {"a": 2, "items": [3], "keep": "yes"}


def test_recursive_dict_merge() -> None:
    base = {
        "database": {"host": "localhost", "port": 5432, "options": {"timeout": 10}},
        "debug": False,
    }
    override = {
        "database": {"port": 5433, "options": {"retries": 3}},
        "log_level": "info",
    }
    assert deep_merge(base, override) == {
        "database": {
            "host": "localhost",
            "port": 5433,
            "options": {"timeout": 10, "retries": 3},
        },
        "debug": False,
        "log_level": "info",
    }


def test_type_mismatch_between_dict_and_non_dict() -> None:
    base = {"a": {"nested": 1}, "b": "scalar"}
    override = {"a": "scalar", "b": {"nested": 2}}
    assert deep_merge(base, override) == {"a": "scalar", "b": {"nested": 2}}


def test_empty_dicts() -> None:
    assert deep_merge({}, {}) == {}
    assert deep_merge({"a": 1}, {}) == {"a": 1}
    assert deep_merge({}, {"a": 1}) == {"a": 1}


def test_inputs_are_not_mutated() -> None:
    base = {"nested": {"a": 1, "b": 2}, "top": "value"}
    override = {"nested": {"b": 3, "c": 4}, "extra": True}
    base_snapshot = deepcopy(base)
    override_snapshot = deepcopy(override)

    result = deep_merge(base, override)

    assert base == base_snapshot
    assert override == override_snapshot
    assert result is not base
    assert result is not override
    assert result["nested"] is not base["nested"]
    assert result["nested"] is not override["nested"]
```
