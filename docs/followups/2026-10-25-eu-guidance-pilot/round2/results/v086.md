"""Configuration reader module."""

from pathlib import Path
import tomllib


def read_config(path: str | Path) -> tuple[int, int]:
    """Read a TOML file and return its timeout and retries values.

    Args:
        path: Path to the TOML configuration file.

    Returns:
        A tuple containing (timeout, retries) as integers.
    """
    with open(path, "rb") as file_handle:
        data = tomllib.load(file_handle)

    return int(data["timeout"]), int(data["retries"])

"""Tests for the config reader module."""

from pathlib import Path
import pytest

from config import read_config


def test_read_config_valid(tmp_path: Path) -> None:
    """Test reading a valid configuration file."""
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 10\nretries = 3\n", encoding="utf-8")

    result = read_config(config_file)
    assert result == (10, 3)
    assert isinstance(result[0], int)
    assert isinstance(result[1], int)
