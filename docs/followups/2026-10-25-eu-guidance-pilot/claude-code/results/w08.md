"""Read the `timeout` and `retries` settings from a TOML file."""

import tomllib


def _require_int(data, key):
    """Return data[key] as an int, or raise a clear error."""
    if key not in data:
        raise KeyError(f"missing required key: {key!r}")
    value = data[key]
    # bool is a subclass of int, but `timeout = true` is almost certainly a mistake.
    if isinstance(value, bool) or not isinstance(value, int):
        raise ValueError(
            f"key {key!r} must be an integer, got {type(value).__name__}: {value!r}"
        )
    return value


def read_config(path):
    """Read a TOML file and return (timeout, retries) as a tuple of two ints.

    Both keys must be set at the top level of the file. A missing key raises
    KeyError, and a value that is not an integer raises ValueError. A file that
    is not valid TOML raises tomllib.TOMLDecodeError, and a missing file raises
    FileNotFoundError.
    """
    with open(path, "rb") as handle:
        data = tomllib.load(handle)
    return _require_int(data, "timeout"), _require_int(data, "retries")

import tomllib

import pytest

from config import read_config


def write(tmp_path, text, name="settings.toml"):
    path = tmp_path / name
    path.write_text(text, encoding="utf-8")
    return path


def test_reads_both_values(tmp_path):
    path = write(tmp_path, "timeout = 30\nretries = 5\n")
    assert read_config(path) == (30, 5)


def test_returns_tuple_of_ints(tmp_path):
    path = write(tmp_path, "timeout = 30\nretries = 5\n")
    result = read_config(path)
    assert isinstance(result, tuple)
    assert len(result) == 2
    assert all(type(value) is int for value in result)


def test_accepts_string_path(tmp_path):
    path = write(tmp_path, "timeout = 10\nretries = 2\n")
    assert read_config(str(path)) == (10, 2)


def test_ignores_extra_keys_and_comments(tmp_path):
    text = (
        "# service settings\n"
        'name = "demo"\n'
        "timeout = 15  # seconds\n"
        "retries = 0\n"
        "\n"
        "[other]\n"
        "timeout = 999\n"
        "retries = 999\n"
    )
    path = write(tmp_path, text)
    assert read_config(path) == (15, 0)


def test_zero_and_large_values(tmp_path):
    path = write(tmp_path, "timeout = 0\nretries = 1_000_000\n")
    assert read_config(path) == (0, 1000000)


def test_missing_timeout_raises_key_error(tmp_path):
    path = write(tmp_path, "retries = 3\n")
    with pytest.raises(KeyError, match="timeout"):
        read_config(path)


def test_missing_retries_raises_key_error(tmp_path):
    path = write(tmp_path, "timeout = 3\n")
    with pytest.raises(KeyError, match="retries"):
        read_config(path)


def test_keys_inside_a_table_are_not_top_level(tmp_path):
    path = write(tmp_path, "[server]\ntimeout = 3\nretries = 1\n")
    with pytest.raises(KeyError):
        read_config(path)


@pytest.mark.parametrize(
    "text",
    [
        'timeout = "30"\nretries = 5\n',
        "timeout = 30\nretries = 2.5\n",
        "timeout = true\nretries = 5\n",
        "timeout = 30\nretries = [1, 2]\n",
    ],
)
def test_non_integer_value_raises_value_error(tmp_path, text):
    path = write(tmp_path, text)
    with pytest.raises(ValueError):
        read_config(path)


def test_invalid_toml_raises_decode_error(tmp_path):
    path = write(tmp_path, "timeout = = 30\n")
    with pytest.raises(tomllib.TOMLDecodeError):
        read_config(path)


def test_missing_file_raises_file_not_found(tmp_path):
    with pytest.raises(FileNotFoundError):
        read_config(tmp_path / "does_not_exist.toml")
