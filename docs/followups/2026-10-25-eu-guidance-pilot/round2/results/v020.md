#!/usr/bin/env python3
"""Command-line hook script to inspect tool call payloads."""

import json
import sys


def evaluate_payload(raw_payload: str) -> str:
    """Determine whether the tool call payload should be allowed or denied.

    Parameters
    ----------
    raw_payload : str
        The raw JSON text received from standard input.

    Returns
    -------
    str
        "deny" if the command contains the forbidden text,
        otherwise "allow".
    """
    if not raw_payload or not raw_payload.strip():
        return "allow"

    try:
        data = json.loads(raw_payload)
        if isinstance(data, dict):
            command = data.get("command")
            if isinstance(command, str):
                if "rm -rf /" in command:
                    return "deny"
                return "allow"
    except Exception:
        pass

    # If the payload cannot be parsed as valid JSON (for example when truncated),
    # inspect the raw payload directly for the forbidden command text.
    if "rm -rf /" in raw_payload:
        return "deny"

    return "allow"


def main() -> None:
    """Read input payload from stdin and output the decision JSON."""
    try:
        raw_payload = sys.stdin.read()
    except Exception:
        raw_payload = ""

    decision = evaluate_payload(raw_payload)
    output = json.dumps({"decision": decision})
    print(output, flush=True)
    sys.exit(0)


if __name__ == "__main__":
    main()

"""Unit and integration tests for hook.py."""

import json
import subprocess
import sys
from pathlib import Path
import pytest
from hook import evaluate_payload

HOOK_SCRIPT = Path(__file__).parent / "hook.py"


def run_hook_process(stdin_content: str) -> subprocess.CompletedProcess[str]:
    """Execute hook.py with the provided standard input content."""
    return subprocess.run(
        [sys.executable, str(HOOK_SCRIPT)],
        input=stdin_content,
        text=True,
        capture_output=True,
        check=False,
    )


class TestEvaluatePayload:
    """Unit tests for the evaluate_payload function."""

    def test_allow_standard_command(self) -> None:
        payload = json.dumps({"tool": "Bash", "command": "ls -la"})
        assert evaluate_payload(payload) == "allow"

    def test_deny_exact_rm_rf_slash(self) -> None:
        payload = json.dumps({"tool": "Bash", "command": "rm -rf /"})
        assert evaluate_payload(payload) == "deny"

    def test_deny_command_containing_rm_rf_slash(self) -> None:
        payload = json.dumps({"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"})
        assert evaluate_payload(payload) == "deny"

    def test_deny_command_with_subpath(self) -> None:
        payload = json.dumps({"tool": "Bash", "command": "rm -rf /tmp/test"})
