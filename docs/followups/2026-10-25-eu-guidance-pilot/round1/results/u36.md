from copy import deepcopy
from typing import Any


def deep_merge(base: dict[str, Any], override: dict[str, Any]) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for key, value in base.items():
        if key not in override:
            result[key] = deepcopy(value)
    for key, override_value in override.items():
        if (
            key in base
            and isinstance(base[key], dict)
            and isinstance(override_value, dict)
        ):
            result[key] = deep_merge(base[key], override_value)
        else:
            result[key] = deepcopy(override_value)
    return result

from merge import deep_merge


def test_disjoint_keys() -> None:
    base = {"a": 1}
    override = {"b": 2}
    assert deep_merge(base, override) == {"a": 1, "b": 2}


def test_override_scalar() -> None:
    base = {"a": 1, "b": 2}
    override = {"b": 3}
    assert deep_merge(base, override) == {"a": 1, "b": 3}


def test_recursive_dict_merge() -> None:
    base = {"nested": {"a": 1, "b": {"x": 10, "y": 20}}, "keep": True}
    override = {"nested": {"b": {"y": 99, "z": 30}, "c": 3}}
    assert deep_merge(base, override) == {
        "nested": {"a": 1, "b": {"x": 10, "y": 99, "z": 30}, "c": 3},
        "keep": True,
    }


def test_dict_and_non_dict_conflict() -> None:
    assert deep_merge({"a": {"nested": 1}}, {"a": 42}) == {"a": 42}
    assert deep_merge({"a": 42}, {"a": {"nested": 1}}) == {"a": {"nested": 1}}
    assert deep_merge({"a": [1, 2]}, {"a": [3]}) == {"a": [3]}


def test_does_not_mutate_inputs() -> None:
    base = {"nested": {"a": 1}, "items": [1, 2]}
    override = {"nested": {"b": 2}, "other": [3, 4]}
    merged = deep_merge(base, override)

    merged["nested"]["a"] = 99
    merged["nested"]["b"] = 99
    merged["items"].append(99)
    merged["other"].append(99)

    assert base == {"nested": {"a": 1}, "items": [1, 2]}
    assert override == {"nested": {"b": 2}, "other": [3, 4]}
