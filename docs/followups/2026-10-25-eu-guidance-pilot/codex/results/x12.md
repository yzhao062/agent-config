import json
from pathlib import Path
import subprocess
import sys

import pytest


HOOK = Path(__file__).with_name("hook.py")


@pytest.mark.parametrize(
    ("payload", "expected"),
    [
        ('{"tool": "Bash", "command": "ls -la"}', "allow"),
        ('{"tool": "Bash", "command": "rm -rf /"}', "deny"),
        ('{"command": "sudo rm -rf /tmp/example"}', "deny"),
        ('{"command": "rm -rf relative-directory"}', "allow"),
        ("{}", "allow"),
        ("", "allow"),
        ("   \n", "allow"),
        ('{"tool": "Bash", "command":', "allow"),
        ('{"command": "ls', "allow"),
        ('{"command": null}', "allow"),
        ("null", "allow"),
    ],
)
def test_hook(payload, expected):
    result = subprocess.run(
        [sys.executable, str(HOOK)],
        input=payload,
        capture_output=True,
        text=True,
        check=False,
    )

    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": expected}
    assert result.stderr == ""
