"""Module providing deep merge functionality for dictionaries."""


def deep_merge(base: dict, override: dict) -> dict:
    """Return a new dictionary that recursively merges base and override.

    For any key present in both dictionaries where both values are dictionaries,
    the values are merged recursively. Otherwise, the value from override takes
    precedence. Neither input dictionary is mutated.
    """
    result = dict(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

"""Unit tests for the deep_merge function in merge.py."""

import copy

import pytest

from merge import deep_merge


def test_both_empty() -> None:
    assert deep_merge({}, {}) == {}


def test_base_empty() -> None:
    override = {"key": "value", "count": 10}
    assert deep_merge({}, override) == {"key": "value", "count": 10}


def test_override_empty() -> None:
    base = {"key": "value", "count": 10}
    assert deep_merge(base, {}) == {"key": "value", "count": 10}


def test_disjoint_keys() -> None:
    base = {"alpha": 1, "beta": 2}
    override = {"gamma": 3, "delta": 4}
    expected = {"alpha": 1, "beta": 2, "gamma": 3, "delta": 4}
    assert deep_merge(base, override) == expected
