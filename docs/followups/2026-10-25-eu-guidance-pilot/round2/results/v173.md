"""Count lines in files specified on the command line."""

import sys


def count_lines(filepath: str) -> int:
    """Return the number of lines in the specified file."""
    with open(filepath, "r", encoding="utf-8", errors="replace") as file:
        return sum(1 for _ in file)


def main(arguments: list[str] | None = None) -> None:
    """Print the line count and name for each file argument."""
    if arguments is None:
        arguments = sys.argv[1:]

    for filepath in arguments:
        line_count = count_lines(filepath)
        print(f"{line_count} {filepath}")


if __name__ == "__main__":
    main()
