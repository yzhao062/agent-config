Design notes:
- An empty, truncated, or non-JSON payload prints `allow`, because the host blocks the call when the hook prints nothing or exits non-zero.
- If a payload does not parse but its raw text contains `rm -rf /`, the hook still prints `deny`. A truncated payload could otherwise hide a dangerous command.
- `rm -rf /tmp/x` also contains the text `rm -rf /`, so the hook denies it. The task defines the rule as a text match, so I kept it.

FILE: hook.py
```python
import json
import sys

DANGEROUS = "rm -rf /"


def decide(raw):
    """Return "deny" or "allow" for a raw stdin payload.

    A malformed, empty, or truncated payload must not block the tool call,
    so it is allowed, unless the raw text itself contains the dangerous text.
    """
    try:
        data = json.loads(raw)
    except ValueError:
        return "deny" if DANGEROUS in raw else "allow"
    command = data.get("command") if isinstance(data, dict) else None
    if isinstance(command, str) and DANGEROUS in command:
        return "deny"
    return "allow"


def main():
    try:
        raw = sys.stdin.read()
    except Exception:
        raw = ""
    print(json.dumps({"decision": decide(raw)}))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
import json
import subprocess
import sys
from pathlib import Path

import pytest

HOOK = str(Path(__file__).with_name("hook.py"))


def run(stdin):
    proc = subprocess.run(
        [sys.executable, HOOK], input=stdin, capture_output=True, text=True
    )
    assert proc.returncode == 0
    return json.loads(proc.stdout)["decision"]


@pytest.mark.parametrize(
    "payload, expected",
    [
        ({"tool": "Bash", "command": "ls -la"}, "allow"),
        ({"tool": "Bash", "command": "rm -rf /"}, "deny"),
        ({"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"}, "deny"),
        ({"tool": "Read"}, "allow"),
        ({"tool": "Bash", "command": None}, "allow"),
    ],
)
def test_valid_payloads(payload, expected):
    assert run(json.dumps(payload)) == expected


@pytest.mark.parametrize(
    "stdin", ["", "   \n", '{"tool": "Bash", "comm', "not json", "[1, 2]", "null"]
)
def test_bad_payloads_are_allowed(stdin):
    assert run(stdin) == "allow"


def test_truncated_payload_with_dangerous_text_is_denied():
    assert run('{"tool": "Bash", "command": "rm -rf /') == "deny"
```
