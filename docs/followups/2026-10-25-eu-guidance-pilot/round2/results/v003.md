"""Module providing dictionary merge utilities."""


def deep_merge(base: dict, override: dict) -> dict:
    """Merge two dictionaries recursively without mutating either input.

    For a key present in both dictionaries where both values are dictionaries,
    the nested dictionaries are merged recursively. Otherwise, the value from
    the override dictionary takes precedence.

    Parameters
    ----------
    base : dict
        The base dictionary.
    override : dict
        The dictionary containing overrides.

    Returns
    -------
    dict
        A new dictionary containing the merged key-value pairs.
    """
    result = dict(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

import copy
from merge import deep_merge


def test_both_empty() -> None:
    """Merging two empty dictionaries should return an empty dictionary."""
    base: dict = {}
    override: dict = {}
    result = deep_merge(base, override)
    assert result == {}
    assert result is not base
    assert result is not override


def test_base_empty() -> None:
    """Merging into an empty base dictionary should return override contents."""
    base: dict = {}
    override = {"a": 1, "b": 2}
    result = deep_merge(base, override)
    assert result == {"a": 1, "b": 2}
    assert result is not override


def test_override_empty() -> None:
    """Merging an empty override dictionary should preserve base contents."""
    base = {"a": 1, "b": 2}
    override: dict = {}
    result = deep_merge(base, override)
    assert result == {"a": 1, "b": 2}
    assert result is not base


def test_disjoint_keys() -> None:
    """Keys present in only one dictionary should all be preserved."""
    base = {"a": 1, "b": 2}
    override = {"c": 3, "d": 4}
    result = deep_merge(base, override)
    assert result == {"a": 1, "b": 2, "c": 3, "d": 4}

    """Scalar values in override should overwrite scalar values in base."""

    result = deep_merge(base, override)

            "host": "localhost",
