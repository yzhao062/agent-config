```python
"""Read timeout and retries settings from a TOML file."""

import tomllib


def read_config(path):
    """Return (timeout, retries) as a tuple of two ints from the TOML file at path."""
    with open(path, "rb") as f:
        data = tomllib.load(f)
    return int(data["timeout"]), int(data["retries"])
```

```python
import pytest

from config import read_config


def write(tmp_path, text):
    p = tmp_path / "c.toml"
    p.write_text(text, encoding="utf-8")
    return p


def test_reads_values(tmp_path):
    p = write(tmp_path, "timeout = 30\nretries = 3\n")
    assert read_config(p) == (30, 3)


def test_returns_ints(tmp_path):
    result = read_config(write(tmp_path, "timeout = 5\nretries = 1\n"))
    assert all(isinstance(v, int) for v in result)


def test_missing_key(tmp_path):
    with pytest.raises(KeyError):
        read_config(write(tmp_path, "timeout = 5\n"))


def test_missing_file(tmp_path):
    with pytest.raises(FileNotFoundError):
        read_config(tmp_path / "nope.toml")


def test_invalid_toml(tmp_path):
    import tomllib

    with pytest.raises(tomllib.TOMLDecodeError):
        read_config(write(tmp_path, "timeout = = 5"))
```
