from copy import deepcopy


def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge two dictionaries and return a new dictionary."""
    merged = deepcopy(base)
    for key, override_value in override.items():
        base_value = merged.get(key)
        if isinstance(base_value, dict) and isinstance(override_value, dict):
            merged[key] = deep_merge(base_value, override_value)
        else:
            merged[key] = deepcopy(override_value)
    return merged

from copy import deepcopy

from merge import deep_merge


def test_disjoint_keys() -> None:
    base = {"host": "localhost"}
    override = {"port": 8080}

    result = deep_merge(base, override)

    assert result == {"host": "localhost", "port": 8080}


def test_override_replaces_scalar_and_list_values() -> None:
    base = {"timeout": 10, "tags": ["a", "b"], "enabled": True}
    override = {"timeout": 30, "tags": ["c"], "enabled": False}

    result = deep_merge(base, override)

    assert result == {"timeout": 30, "tags": ["c"], "enabled": False}


def test_nested_dicts_merge_recursively() -> None:
    base = {
        "database": {
            "host": "localhost",
            "port": 5432,
            "options": {"ssl": True, "pool_size": 5},
        },
        "logging": {"level": "INFO"},
    }

        "database": {
            "port": 6432,
            "options": {"pool_size": 20, "timeout": 15},

        "feature_flags": {"new_ui": True},

    result = deep_merge(base, override)

    assert result == {
        "database": {
            "host": "localhost",
            "port": 6432,
            "options": {"ssl": True, "pool_size": 20, "timeout": 15},
        },
        "logging": {"level": "INFO"},
        "feature_flags": {"new_ui": True},
    }


def test_mismatched_dict_and_non_dict_types_use_override() -> None:
    base = {"section": {"nested": 1}, "scalar": "text"}
    override = {"section": "replaced", "scalar": {"nested": 2}}

    result = deep_merge(base, override)

    assert result == {"section": "replaced", "scalar": {"nested": 2}}


def test_empty_dicts() -> None:
    sample = {"a": {"b": 1}}

    assert deep_merge({}, {}) == {}
    assert deep_merge(sample, {}) == {"a": {"b": 1}}
    assert deep_merge({}, sample) == {"a": {"b": 1}}


def test_inputs_are_not_mutated() -> None:
    base = {"nested": {"keep": 1, "change": 2}, "items": [1, 2]}
    override = {"nested": {"change": 3, "add": 4}, "extra": [5]}
    base_snapshot = deepcopy(base)
    override_snapshot = deepcopy(override)

    result = deep_merge(base, override)
    result["nested"]["keep"] = 99
    result["items"].append(3)
    result["extra"].append(6)

    assert base == base_snapshot
    assert override == override_snapshot
