```python
from pathlib import Path
import tomllib


def read_config(path: str | Path) -> tuple[int, int]:
    with open(path, "rb") as f:
        data = tomllib.load(f)
    return int(data["timeout"]), int(data["retries"])
```

```python
from pathlib import Path
import pytest
from config import read_config


def test_read_config_returns_timeout_and_retries(tmp_path: Path) -> None:
    cfg = tmp_path / "config.toml"
    cfg.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")
    assert read_config(cfg) == (30, 3)


def test_read_config_missing_key_raises(tmp_path: Path) -> None:
    cfg = tmp_path / "config.toml"
    cfg.write_text("timeout = 30\n", encoding="utf-8")
    with pytest.raises(KeyError):
        read_config(cfg)
```
