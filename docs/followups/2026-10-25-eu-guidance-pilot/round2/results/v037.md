```python
"""Configuration reader module."""

from pathlib import Path
import tomllib


def read_config(path: str | Path) -> tuple[int, int]:
    """Read a TOML file and return timeout and retries values as a tuple of two ints."""
    with open(path, "rb") as file:
        data = tomllib.load(file)
    return int(data["timeout"]), int(data["retries"])
```

```python
"""Tests for the configuration reader module."""

from pathlib import Path
import pytest
from config import read_config


def test_read_config_returns_timeout_and_retries(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 30\nretries = 5\n", encoding="utf-8")

    result = read_config(config_file)

    assert result == (30, 5)
    assert isinstance(result[0], int)
    assert isinstance(result[1], int)


def test_read_config_with_string_path(tmp_path: Path) -> None:
    config_file = tmp_path / "settings.toml"
    config_file.write_text("timeout = 10\nretries = 2\n", encoding="utf-8")

    result = read_config(str(config_file))

    assert result == (10, 2)


def test_read_config_with_extra_fields(tmp_path: Path) -> None:
    config_file = tmp_path / "full_config.toml"
    content = (
        "name = \"service\"\n"
        "timeout = 60\n"
        "retries = 4\n"
        "debug = true\n"
    )
    config_file.write_text(content, encoding="utf-8")

    result = read_config(config_file)

    assert result == (60, 4)
```
