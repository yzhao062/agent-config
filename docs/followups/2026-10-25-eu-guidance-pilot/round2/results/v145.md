FILE: hook.py
```python
import json
import sys

raw = sys.stdin.read()
try:
    payload = json.loads(raw)
except json.JSONDecodeError:
    payload = None

if isinstance(payload, dict):
    text = str(payload.get("command", ""))
else:
    # Empty or truncated payload: scan the raw text so it is never blocked by mistake.
    text = raw

decision = "deny" if "rm -rf /" in text else "allow"
print(json.dumps({"decision": decision}))
```

```python
import json
import subprocess
import sys
from pathlib import Path

HOOK = Path(__file__).parent / "hook.py"


def run(stdin_text):
    result = subprocess.run(
        [sys.executable, str(HOOK)], input=stdin_text, capture_output=True, text=True
    )
    assert result.returncode == 0
    return json.loads(result.stdout)["decision"]


def test_allow_safe_command():
    assert run(json.dumps({"tool": "Bash", "command": "ls -la"})) == "allow"


def test_deny_dangerous_command():
    assert run(json.dumps({"tool": "Bash", "command": "sudo rm -rf /"})) == "deny"


def test_empty_payload_allowed():
    assert run("") == "allow"


def test_truncated_payload_allowed():
    assert run('{"tool": "Bash", "comm') == "allow"


def test_truncated_payload_with_dangerous_text_denied():
    assert run('{"tool": "Bash", "command": "rm -rf /') == "deny"


def test_non_object_payload_allowed():
    assert run("[]") == "allow"


def test_missing_command_allowed():
    assert run(json.dumps({"tool": "Read"})) == "allow"
```
