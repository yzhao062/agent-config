"""Module providing deep merge functionality for dictionaries."""


def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge override dictionary into base dictionary.

    For keys present in both dictionaries where both values are dictionaries,
    merge them recursively. Otherwise, the value from override is used.
    Neither input dictionary is mutated.
    """
    result = dict(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

"""Unit tests for the deep_merge function."""

from merge import deep_merge


def test_merge_empty_dictionaries():
    assert deep_merge({}, {}) == {}
    assert deep_merge({"a": 1}, {}) == {"a": 1}
    assert deep_merge({}, {"a": 1}) == {"a": 1}


def test_merge_disjoint_keys():
    base = {"a": 1}
    override = {"b": 2}
    expected = {"a": 1, "b": 2}
    assert deep_merge(base, override) == expected


def test_override_flat_values():
    base = {"a": 1, "b": 2}
    override = {"b": 99, "c": 3}
    expected = {"a": 1, "b": 99, "c": 3}
    assert deep_merge(base, override) == expected


def test_override_with_different_types():
    base = {"a": {"nested": 1}, "b": 2}
    override = {"a": "replaced", "b": {"nested": 3}}
    expected = {"a": "replaced", "b": {"nested": 3}}
    assert deep_merge(base, override) == expected


def test_deep_merge_nested_dictionaries():
    base = {
        "level1": {
            "level2": {
                "keep": True,
                "change": 1,
            },
            "untouched": "yes",
        }
    }
    override = {
        "level1": {
            "level2": {
                "change": 2,
                "add": "new",
            }
        }
    }
    expected = {
        "level1": {
            "level2": {
                "keep": True,
                "change": 2,
                "add": "new",
            },
            "untouched": "yes",
        }
    }
    assert deep_merge(base, override) == expected


def test_deep_merge_does_not_mutate_inputs():
    base = {"a": 1, "nested": {"x": 10}}
    override = {"b": 2, "nested": {"y": 20}}

    base_copy = {"a": 1, "nested": {"x": 10}}
    override_copy = {"b": 2, "nested": {"y": 20}}

    result = deep_merge(base, override)

    assert base == base_copy
    assert override == override_copy
    assert result is not base
    assert result is not override
    assert result["nested"] is not base["nested"]
    assert result["nested"] is not override["nested"]
