"""Deep merge helper for settings dictionaries."""

import copy


def deep_merge(base: dict, override: dict) -> dict:
    """Return a new dict that merges override into base recursively.

    When a key is in both dicts and both values are dicts, the values are
    merged recursively. Otherwise the value from override wins. Neither input
    is mutated, and the result shares no mutable values with the inputs.
    """
    result = {}
    for key, value in base.items():
        if key in override:
            continue
        result[key] = copy.deepcopy(value)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(base.get(key), dict):
            result[key] = deep_merge(base[key], value)
        else:
            result[key] = copy.deepcopy(value)
    return result

from merge import deep_merge


def test_disjoint_keys():
    assert deep_merge({"a": 1}, {"b": 2}) == {"a": 1, "b": 2}


def test_override_wins_for_scalars():
    assert deep_merge({"a": 1}, {"a": 2}) == {"a": 2}


def test_nested_merge():
    base = {"a": {"x": 1, "y": 2}, "b": 1}
    override = {"a": {"y": 3, "z": 4}}
    assert deep_merge(base, override) == {"a": {"x": 1, "y": 3, "z": 4}, "b": 1}


def test_dict_replaced_by_non_dict_and_back():
    assert deep_merge({"a": {"x": 1}}, {"a": 5}) == {"a": 5}
    assert deep_merge({"a": 5}, {"a": {"x": 1}}) == {"a": {"x": 1}}


def test_lists_are_replaced():
    assert deep_merge({"a": [1, 2]}, {"a": [3]}) == {"a": [3]}


def test_none_override_wins():
    assert deep_merge({"a": 1}, {"a": None}) == {"a": None}


def test_empty_inputs():
    assert deep_merge({}, {}) == {}
    assert deep_merge({"a": 1}, {}) == {"a": 1}
    assert deep_merge({}, {"a": 1}) == {"a": 1}


def test_inputs_not_mutated_and_not_shared():
    base = {"a": {"x": [1]}, "k": {"n": 1}}
    override = {"a": {"y": 2}, "o": {"m": 1}}
    base_copy = {"a": {"x": [1]}, "k": {"n": 1}}
    override_copy = {"a": {"y": 2}, "o": {"m": 1}}
    result = deep_merge(base, override)
    assert base == base_copy
    assert override == override_copy
    result["a"]["x"].append(9)
    result["k"]["n"] = 7
    result["o"]["m"] = 7
    assert base == base_copy
    assert override == override_copy
