```python
import tomllib


def read_config(path):
    """Read a TOML file and return (timeout, retries) as a tuple of two ints.

    Raises KeyError if a key is missing and ValueError if a value is not an
    integer (a float timeout such as 2.5 is rejected rather than truncated).
    """
    with open(path, "rb") as f:
        data = tomllib.load(f)

    values = []
    for key in ("timeout", "retries"):
        value = data[key]
        if type(value) is not int:
            raise ValueError(f"{key} must be an integer, got {value!r}")
        values.append(value)
    return tuple(values)
```

```python
import pytest

from config import read_config


def write(tmp_path, text):
    path = tmp_path / "config.toml"
    path.write_text(text)
    return path


def test_reads_timeout_and_retries(tmp_path):
    path = write(tmp_path, "timeout = 30\nretries = 3\n")
    assert read_config(path) == (30, 3)


def test_ignores_other_keys(tmp_path):
    path = write(tmp_path, 'name = "svc"\ntimeout = 5\nretries = 0\n[extra]\nx = 1\n')
    assert read_config(path) == (5, 0)


def test_missing_key_raises(tmp_path):
    path = write(tmp_path, "timeout = 30\n")
    with pytest.raises(KeyError):
        read_config(path)


def test_float_value_is_rejected_not_truncated(tmp_path):
    path = write(tmp_path, "timeout = 2.5\nretries = 3\n")
    with pytest.raises(ValueError):
        read_config(path)
```
