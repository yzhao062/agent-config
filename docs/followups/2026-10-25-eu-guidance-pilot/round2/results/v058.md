```python
import copy


def deep_merge(base: dict, override: dict) -> dict:
    """Return a new dict with override merged into base, recursing into nested dicts."""
    result = {}
    for key, value in base.items():
        result[key] = copy.deepcopy(value)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(result.get(key), dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = copy.deepcopy(value)
    return result
```

```python
import copy

from merge import deep_merge


def test_disjoint_keys():
    assert deep_merge({"a": 1}, {"b": 2}) == {"a": 1, "b": 2}


def test_override_wins_for_scalars():
    assert deep_merge({"a": 1}, {"a": 2}) == {"a": 2}


def test_nested_dicts_merge_recursively():
    base = {"a": {"x": 1, "y": {"p": 1, "q": 2}}}
    override = {"a": {"y": {"q": 3}, "z": 4}}
    assert deep_merge(base, override) == {"a": {"x": 1, "y": {"p": 1, "q": 3}, "z": 4}}


def test_non_dict_replaces_dict_and_reverse():
    assert deep_merge({"a": {"x": 1}}, {"a": 5}) == {"a": 5}
    assert deep_merge({"a": 5}, {"a": {"x": 1}}) == {"a": {"x": 1}}


def test_lists_are_replaced():
    assert deep_merge({"a": [1, 2]}, {"a": [3]}) == {"a": [3]}


def test_inputs_not_mutated_and_result_not_aliased():
    base = {"a": {"x": 1}, "l": [1]}
    override = {"a": {"y": 2}, "n": {"k": [1]}}
    base_copy = copy.deepcopy(base)
    override_copy = copy.deepcopy(override)
    result = deep_merge(base, override)
    assert base == base_copy
    assert override == override_copy
    result["a"]["x"] = 99
    result["l"].append(2)
    result["n"]["k"].append(2)
    assert base == base_copy
    assert override == override_copy
```
