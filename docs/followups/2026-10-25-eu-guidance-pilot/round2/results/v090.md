"""Module for merging dictionary structures."""


def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge override into base, returning a new dictionary.

    For keys present in both dictionaries where both values are dictionaries,
    merge them recursively. Otherwise, the value from override is used.
    Neither input dictionary is mutated.
    """
    result = dict(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

"""Tests for the deep_merge function."""

import copy

from merge import deep_merge


def test_empty_dictionaries() -> None:
    """Test merging with empty dictionaries."""
    assert deep_merge({}, {}) == {}
    assert deep_merge({"a": 1}, {}) == {"a": 1}
    assert deep_merge({}, {"b": 2}) == {"b": 2}


def test_flat_merge() -> None:
    """Test merging flat dictionaries with overlapping and non-overlapping keys."""
    base = {"a": 1, "b": 2}
    override = {"b": 3, "c": 4}
    expected = {"a": 1, "b": 3, "c": 4}
    assert deep_merge(base, override) == expected


def test_nested_merge() -> None:
    """Test recursive merging of nested dictionaries."""
    base = {
        "database": {"host": "localhost", "port": 5432},
        "logging": {"level": "INFO"},
    }
    override = {
        "database": {"port": 5433, "name": "production"},
        "logging": {"file": "app.log"},
    }
    expected = {
        "database": {"host": "localhost", "port": 5433, "name": "production"},
        "logging": {"level": "INFO", "file": "app.log"},
    }
    assert deep_merge(base, override) == expected


def test_deeply_nested_merge() -> None:
    """Test merging across multiple levels of nesting."""
    base = {"level1": {"level2": {"level3": {"key1": "val1", "key2": "val2"}}}}
    override = {"level1": {"level2": {"level3": {"key2": "new_val2", "key4": "val4"}}}}
    expected = {
        "level1": {"level2": {"level3": {"key1": "val1", "key2": "new_val2", "key4": "val4"}}}
    }
    assert deep_merge(base, override) == expected


def test_override_non_dict_replaces_dict() -> None:
    """Test that a non-dict override replaces a dictionary in base."""
    base = {"key": {"sub_key": "val"}}

    assert deep_merge(base, override) == {"key": "plain_string"}

    """Test that a dict override replaces a non-dict in base."""
    base = {"key": "plain_string"}

    assert deep_merge(base, override) == {"key": {"sub_key": "val"}}

    override = {"items": [3, 4], "flag": False, "count": None}

    assert deep_merge(base, override) == expected

    override = {"a": 2, "nested": {"y": 30, "z": 40}}

    result = deep_merge(base, override)

    assert base == base_copy
    assert override == override_copy

    result["nested"]["y"] = 999
    assert base["nested"]["y"] == 20
    assert override["nested"]["y"] == 30


def test_returns_new_dictionary_instance() -> None:
    """Test that deep_merge returns a new dictionary object."""
    base = {"a": 1}
    override: dict = {}
    result = deep_merge(base, override)
