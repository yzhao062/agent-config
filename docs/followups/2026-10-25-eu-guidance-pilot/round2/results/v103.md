import json
import sys


def evaluate_payload(raw_payload: str) -> str:
    """Return 'allow' or 'deny' for a JSON hook payload string.

    Fails closed ('deny') when the payload is empty, truncated, malformed,
    or lacks a string 'command' field for a command execution tool, because a
    truncated payload could omit the dangerous portion of a command.
    """
    if not raw_payload or not raw_payload.strip():
        return "deny"

    try:
        payload = json.loads(raw_payload)
    except json.JSONDecodeError:
        return "deny"

    if not isinstance(payload, dict):
        return "deny"

    command = payload.get("command")
    if not isinstance(command, str):
        return "deny"

    if "rm -rf /" in command:
        return "deny"

    return "allow"


def main() -> int:
    raw_payload = sys.stdin.read()
    decision = evaluate_payload(raw_payload)
    sys.stdout.write(json.dumps({"decision": decision}) + "\n")
    sys.stdout.flush()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

import json
import subprocess
import sys
from pathlib import Path

import pytest

from hook import evaluate_payload

HOOK_SCRIPT = Path(__file__).with_name("hook.py")


def run_hook(input_text: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [sys.executable, str(HOOK_SCRIPT)],
        input=input_text,
        capture_output=True,
        text=True,
        check=False,
    )


def test_allows_safe_command() -> None:
    payload = json.dumps({"tool": "Bash", "command": "ls -la"})
    result = run_hook(payload)
    assert result.returncode == 0
    assert result.stdout.strip() == '{"decision": "allow"}'
    assert json.loads(result.stdout) == {"decision": "allow"}


@pytest.mark.parametrize(
    "command",
    [
        "rm -rf /",
        "sudo rm -rf /",
        "echo hello && rm -rf / --no-preserve-root",
        "rm -rf /tmp/test",
    ],
)
def test_denies_command_containing_rm_rf_root(command: str) -> None:
    payload = json.dumps({"tool": "Bash", "command": command})
    result = run_hook(payload)
    assert result.returncode == 0
    assert result.stdout.strip() == '{"decision": "deny"}'
    assert json.loads(result.stdout) == {"decision": "deny"}


@pytest.mark.parametrize(
    "raw_payload",
    [
        "",
        "   \n\t  ",
        '{"tool": "Bash", "command": "ls -la"',
        '{"tool": "Bash", "command": "rm -rf ',
        '{"tool": "Bash"',
        "not json at all",
        "[]",
        '{"tool": "Bash"}',
        '{"tool": "Bash", "command": null}',
        '{"tool": "Bash", "command": 123}',
    ],
)
def test_fails_closed_on_empty_truncated_or_malformed_payload(raw_payload: str) -> None:
    result = run_hook(raw_payload)
    assert result.returncode == 0
    assert result.stdout.strip() == '{"decision": "deny"}'
    assert json.loads(result.stdout) == {"decision": "deny"}


def test_evaluate_payload_unit() -> None:
    assert evaluate_payload('{"tool": "Bash", "command": "pwd"}') == "allow"
    assert evaluate_payload('{"tool": "Bash", "command": "rm -rf /"}') == "deny"
    assert evaluate_payload("") == "deny"
    assert evaluate_payload('{"tool": "Bash", "command": "pwd"') == "deny"
