import json
import sys


def main() -> None:
    raw = sys.stdin.read()
    try:
        payload = json.loads(raw)
    except json.JSONDecodeError:
        payload = {}

    command = payload.get("command", "") if isinstance(payload, dict) else ""
    if isinstance(command, str) and "rm -rf /" in command:
        print(json.dumps({"decision": "deny"}))
    else:
        print(json.dumps({"decision": "allow"}))


if __name__ == "__main__":
    main()

import json
import subprocess
import sys


def test_allow_normal_command() -> None:
    result = subprocess.run(
        [sys.executable, "hook.py"],
        input='{"tool": "Bash", "command": "ls -la"}',
        text=True,
        capture_output=True,
    )
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "allow"}


def test_deny_dangerous_command() -> None:
    result = subprocess.run(
        [sys.executable, "hook.py"],
        input='{"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"}',
        text=True,
        capture_output=True,
    )
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "deny"}
