import json
import sys

payload_text = sys.stdin.read().strip()
command = ""

if payload_text:
    try:
        data = json.loads(payload_text)
        if isinstance(data, dict):
            raw_command = data.get("command")
            if isinstance(raw_command, str):
                command = raw_command
    except json.JSONDecodeError:
        pass

if "rm -rf /" in command:
    print('{"decision": "deny"}')
else:
    print('{"decision": "allow"}')

```python
import pathlib
import subprocess
import sys
import pytest

HOOK_PATH = pathlib.Path(__file__).parent / "hook.py"


@pytest.mark.parametrize(
    ("payload", "expected_stdout"),
    [
        ('{"tool": "Bash", "command": "rm -rf /"}', '{"decision": "deny"}\n'),
        ('{"tool": "Bash", "command": "sudo rm -rf /"}', '{"decision": "deny"}\n'),
        ('{"tool": "Bash", "command": "rm -rf /tmp"}', '{"decision": "deny"}\n'),
        ('{"tool": "Bash", "command": "ls -la"}', '{"decision": "allow"}\n'),
        ('{"tool": "Bash", "command": "echo rm -rf"}', '{"decision": "allow"}\n'),
        ('{"tool": "Bash"}', '{"decision": "allow"}\n'),
        ('{"tool": "Bash", "command": 123}', '{"decision": "allow"}\n'),
        ("", '{"decision": "allow"}\n'),
        ("   \n  ", '{"decision": "allow"}\n'),
        ('{"tool": "Bash", "command": "ls', '{"decision": "allow"}\n'),
        ('{"tool":', '{"decision": "allow"}\n'),
        ("not json", '{"decision": "allow"}\n'),
        ("[]", '{"decision": "allow"}\n'),
    ],
)
def test_hook_decisions(payload: str, expected_stdout: str) -> None:
    process = subprocess.run(
        [sys.executable, str(HOOK_PATH)],
        input=payload,
        text=True,
        capture_output=True,
        check=False,
    )
    assert process.returncode == 0
    assert process.stdout == expected_stdout
```
