from pathlib import Path
import sys


def count_lines(path: Path) -> int:
    with path.open("rb") as f:
        return sum(1 for _ in f)


def main(argv: list[str] | None = None) -> None:
    args = sys.argv[1:] if argv is None else argv
    for name in args:
        count = count_lines(Path(name))
        print(f"{count} {name}")


if __name__ == "__main__":
    main()

from pathlib import Path
import pytest
from count_lines import main


def test_count_lines_multiple_files(tmp_path: Path, capsys: pytest.CaptureFixture[str]) -> None:
    empty = tmp_path / "empty.txt"
    empty.write_text("", encoding="utf-8")

    three = tmp_path / "three.txt"
    three.write_text("first\nsecond\nthird\n", encoding="utf-8")

    main([str(empty), str(three)])
    captured = capsys.readouterr()
    assert captured.out == f"0 {empty}\n3 {three}\n"
