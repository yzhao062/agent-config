import sys


def main() -> None:
    for path in sys.argv[1:]:
        with open(path, "r", encoding="utf-8") as file:
            count = sum(1 for _ in file)
        print(f"{count} {path}")


if __name__ == "__main__":
    main()

import subprocess
import sys
from pathlib import Path

SCRIPT_PATH = Path(__file__).resolve().parent / "count_lines.py"


def test_count_lines_single_file(tmp_path: Path) -> None:
    file_path = tmp_path / "first.txt"
    file_path.write_text("line one\nline two\nline three\n", encoding="utf-8")

    result = subprocess.run(
        [sys.executable, str(SCRIPT_PATH), str(file_path)],
        capture_output=True,
        text=True,
        check=True,
    )
    assert result.stdout == f"3 {file_path}\n"


def test_count_lines_multiple_files(tmp_path: Path) -> None:
    first_file = tmp_path / "a.txt"
    first_file.write_text("hello\n", encoding="utf-8")

    second_file = tmp_path / "b.txt"
    second_file.write_text("line 1\nline 2\n", encoding="utf-8")

    result = subprocess.run(
        [sys.executable, str(SCRIPT_PATH), str(first_file), str(second_file)],
        capture_output=True,
        text=True,
        check=True,
    )
    expected_output = f"1 {first_file}\n2 {second_file}\n"
