"""Module providing dictionary deep merge functionality."""


def deep_merge(base: dict, override: dict) -> dict:
    """Recursively merge two dictionaries, with override values taking precedence.

    For keys present in both dictionaries where both values are dictionaries,
    the values are merged recursively. Otherwise, the value from the override
    dictionary takes precedence. Neither input dictionary is mutated.
    """
    result = dict(base)
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = deep_merge(result[key], value)
        else:
            result[key] = value
    return result

"""Unit tests for the deep_merge function."""

import copy
from merge import deep_merge


def test_merge_disjoint_keys():
    base = {"first": 1}
    override = {"second": 2}
    expected = {"first": 1, "second": 2}
    assert deep_merge(base, override) == expected


def test_merge_override_primitive_wins():
    base = {"key": "old", "unchanged": True}
    override = {"key": "new"}
    expected = {"key": "new", "unchanged": True}
    assert deep_merge(base, override) == expected


def test_merge_both_empty():
    assert deep_merge({}, {}) == {}


def test_merge_empty_base():
    override = {"nested": {"value": 42}}
    assert deep_merge({}, override) == {"nested": {"value": 42}}

    base = {"nested": {"value": 42}}
    assert deep_merge(base, {}) == {"nested": {"value": 42}}
