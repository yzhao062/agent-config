"""Recursive dictionary merge."""

from copy import deepcopy


def deep_merge(base: dict, override: dict) -> dict:
    """Return a new dict with override merged into base.

    When a key is in both dicts and both values are dicts, they are merged
    recursively. Otherwise the value from override wins. Neither input is
    mutated, and the result shares no nested containers with the inputs.
    """
    result = {}
    for key, value in base.items():
        if key in override:
            continue
        result[key] = deepcopy(value)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(base.get(key), dict):
            result[key] = deep_merge(base[key], value)
        else:
            result[key] = deepcopy(value)
    return result

from copy import deepcopy

from merge import deep_merge


def test_disjoint_keys():
    assert deep_merge({"a": 1}, {"b": 2}) == {"a": 1, "b": 2}


def test_override_wins_for_scalars():
    assert deep_merge({"a": 1}, {"a": 2}) == {"a": 2}


def test_nested_dicts_merge_recursively():
    base = {"a": {"x": 1, "y": {"p": 1, "q": 2}}}
    override = {"a": {"y": {"q": 3}, "z": 4}}
    assert deep_merge(base, override) == {"a": {"x": 1, "y": {"p": 1, "q": 3}, "z": 4}}


def test_non_dict_replaces_dict_and_reverse():
    assert deep_merge({"a": {"x": 1}}, {"a": 5}) == {"a": 5}
    assert deep_merge({"a": 5}, {"a": {"x": 1}}) == {"a": {"x": 1}}


def test_lists_are_replaced_not_merged():
    assert deep_merge({"a": [1, 2]}, {"a": [3]}) == {"a": [3]}


def test_none_override_wins():
    assert deep_merge({"a": 1}, {"a": None}) == {"a": None}


def test_inputs_not_mutated_and_result_not_aliased():
    base = {"a": {"x": [1]}, "b": {"k": 1}}
    override = {"a": {"y": 2}, "c": {"m": [2]}}
    base_copy, override_copy = deepcopy(base), deepcopy(override)
    result = deep_merge(base, override)
    assert base == base_copy and override == override_copy
    result["a"]["x"].append(9)
    result["c"]["m"].append(9)
    result["b"]["k"] = 9
    assert base == base_copy and override == override_copy
