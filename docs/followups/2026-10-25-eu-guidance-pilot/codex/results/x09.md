import tomllib


def read_config(path) -> tuple[int, int]:
    with open(path, "rb") as file:
        config = tomllib.load(file)

    timeout = config["timeout"]
    retries = config["retries"]
    if type(timeout) is not int or type(retries) is not int:
        raise TypeError("timeout and retries must be integers")

    return timeout, retries

import tomllib

import pytest

from config import read_config


def test_read_config(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")

    assert read_config(path) == (30, 3)


@pytest.mark.parametrize("key", ["timeout", "retries"])
@pytest.mark.parametrize("value", ["true", "1.5", '"3"', "[3]"])
def test_rejects_non_integer_values(tmp_path, key, value):
    values = {"timeout": "30", "retries": "3"}
    values[key] = value
    path = tmp_path / "config.toml"
    path.write_text(
        "\n".join(f"{name} = {item}" for name, item in values.items()),
        encoding="utf-8",
    )

    with pytest.raises(TypeError, match="must be integers"):
        read_config(path)


@pytest.mark.parametrize("missing", ["timeout", "retries"])
def test_missing_required_value(tmp_path, missing):
    values = {"timeout": 30, "retries": 3}
    del values[missing]
    path = tmp_path / "config.toml"
    path.write_text(
        "\n".join(f"{name} = {value}" for name, value in values.items()),
        encoding="utf-8",
    )

    with pytest.raises(KeyError) as error:
        read_config(path)

    assert error.value.args == (missing,)


def test_invalid_toml(tmp_path):
    path = tmp_path / "config.toml"
    path.write_text("timeout = [\n", encoding="utf-8")

    with pytest.raises(tomllib.TOMLDecodeError):
        read_config(path)


def test_missing_file(tmp_path):
    with pytest.raises(FileNotFoundError):
        read_config(tmp_path / "missing.toml")
