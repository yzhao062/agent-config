"""Print the number of lines in each file named on the command line."""
import sys


def count_lines(path):
    """Return the number of lines in the file at path (binary-safe)."""
    count = 0
    last = b""
    with open(path, "rb") as handle:
        for chunk in iter(lambda: handle.read(65536), b""):
            count += chunk.count(b"\n")
            last = chunk[-1:]
    if last and last != b"\n":
        count += 1  # final line without a trailing newline
    return count


def main(argv=None):
    args = sys.argv[1:] if argv is None else argv
    status = 0
    for name in args:
        try:
            print(f"{count_lines(name)} {name}")
        except OSError as error:
            print(f"count_lines: {name}: {error.strerror}", file=sys.stderr)
            status = 1
    return status


if __name__ == "__main__":
    sys.exit(main())

from count_lines import count_lines, main


def test_trailing_newline(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"one\ntwo\n")
    assert count_lines(str(f)) == 2


def test_no_trailing_newline(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"one\ntwo")
    assert count_lines(str(f)) == 2


def test_empty_file(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"")
    assert count_lines(str(f)) == 0


def test_main_multiple_files(tmp_path, capsys):
    a = tmp_path / "a.txt"
    b = tmp_path / "b.txt"
    a.write_bytes(b"x\n")
    b.write_bytes(b"x\ny\nz\n")
    assert main([str(a), str(b)]) == 0
    assert capsys.readouterr().out == f"1 {a}\n3 {b}\n"


def test_main_missing_file(tmp_path, capsys):
    a = tmp_path / "a.txt"
    a.write_bytes(b"x\n")
    missing = str(tmp_path / "nope")
    assert main([missing, str(a)]) == 1
    captured = capsys.readouterr()
    assert captured.out == f"1 {a}\n"
