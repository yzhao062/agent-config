```python
import tomllib


def read_config(path):
    """Read a TOML file and return (timeout, retries) as ints."""
    with open(path, "rb") as file:
        data = tomllib.load(file)
    return int(data["timeout"]), int(data["retries"])
```

```python
import pytest

from config import read_config


def test_read_config_returns_timeout_and_retries(tmp_path):
    path = tmp_path / "settings.toml"
    path.write_text("timeout = 30\nretries = 3\n")
    assert read_config(path) == (30, 3)


def test_read_config_ignores_other_keys(tmp_path):
    path = tmp_path / "settings.toml"
    path.write_text('name = "demo"\nretries = 5\ntimeout = 10\n')
    assert read_config(path) == (10, 5)


def test_read_config_missing_key_raises(tmp_path):
    path = tmp_path / "settings.toml"
    path.write_text("timeout = 30\n")
    with pytest.raises(KeyError):
        read_config(path)
```
