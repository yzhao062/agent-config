import tomllib


def read_config(path):
    with open(path, "rb") as f:
        data = tomllib.load(f)
    return int(data["timeout"]), int(data["retries"])

import pytest

from config import read_config


def test_reads_values(tmp_path):
    p = tmp_path / "c.toml"
    p.write_text("timeout = 30\nretries = 3\n")
    assert read_config(p) == (30, 3)


def test_ignores_extra_keys(tmp_path):
    p = tmp_path / "c.toml"
    p.write_text('name = "x"\nretries = 5\ntimeout = 10\n')
    assert read_config(p) == (10, 5)


def test_missing_key_raises(tmp_path):
    p = tmp_path / "c.toml"
    p.write_text("timeout = 30\n")
    with pytest.raises(KeyError):
        read_config(p)
