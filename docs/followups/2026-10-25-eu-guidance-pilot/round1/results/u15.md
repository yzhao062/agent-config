from __future__ import annotations

import json
import sys
from typing import TextIO

FORBIDDEN_SUBSTRING = "rm -rf /"
ALLOW_RESPONSE = json.dumps({"decision": "allow"})
DENY_RESPONSE = json.dumps({"decision": "deny"})


def evaluate_payload(raw_payload: str) -> str:
    """Evaluate a JSON payload from the host and return a JSON decision string.

    Empty, truncated, or malformed payloads fail closed ("deny") because the
    command cannot be safely inspected.
    """
    if not raw_payload or not raw_payload.strip():
        return DENY_RESPONSE

    try:
        data = json.loads(raw_payload)
    except json.JSONDecodeError:
        return DENY_RESPONSE

    if not isinstance(data, dict):
        return DENY_RESPONSE

    command = data.get("command")
    if not isinstance(command, str):
        return DENY_RESPONSE

    if FORBIDDEN_SUBSTRING in command:
        return DENY_RESPONSE

    return ALLOW_RESPONSE


def main(stdin: TextIO = sys.stdin, stdout: TextIO = sys.stdout) -> int:
    raw_payload = stdin.read()
    decision = evaluate_payload(raw_payload)
    stdout.write(f"{decision}\n")
    stdout.flush()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

import io
import json
import subprocess
import sys
from pathlib import Path

import pytest

import hook

HOOK_PATH = Path(__file__).with_name("hook.py")


def run_hook_subprocess(payload: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [sys.executable, str(HOOK_PATH)],
        input=payload,
        capture_output=True,
        text=True,
        check=False,
    )


def test_allows_safe_command() -> None:
    payload = json.dumps({"tool": "Bash", "command": "ls -la"})
    result = run_hook_subprocess(payload)
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "allow"}


@pytest.mark.parametrize(
    "command",
    [
        "rm -rf /",
        "sudo rm -rf / --no-preserve-root",
        "echo hello && rm -rf /tmp",
    ],
)
def test_denies_forbidden_command(command: str) -> None:
    payload = json.dumps({"tool": "Bash", "command": command})
    result = run_hook_subprocess(payload)
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "deny"}


@pytest.mark.parametrize(
    "payload",
    [
        "",
        "   \n  ",
        "{",
        '{"tool": "Bash", "command": "ls -la"',
        '{"tool": "Bash", "command": "rm -rf ',
        "null",
        "[]",
        '{"tool": "Bash"}',
        '{"tool": "Bash", "command": null}',
    ],
)
def test_fails_closed_on_empty_truncated_or_malformed_payload(payload: str) -> None:
    result = run_hook_subprocess(payload)
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "deny"}


def test_main_uses_provided_streams() -> None:
    stdin = io.StringIO('{"tool": "Bash", "command": "pwd"}')
    stdout = io.StringIO()
    exit_code = hook.main(stdin=stdin, stdout=stdout)
    assert exit_code == 0
    assert stdout.getvalue() == '{"decision": "allow"}\n'
