import pathlib
import sys


def main(argv: list[str]) -> int:
    if not argv:
        sys.stderr.write("Usage: count_lines.py <file>...\n")
        return 1

    paths = [pathlib.Path(arg) for arg in argv]
    for path in paths:
        if not path.is_file():
            sys.stderr.write(f"Error: {path} is not a file\n")
            return 1

    for arg, path in zip(argv, paths, strict=True):
        with path.open("r", encoding="utf-8") as handle:
            count = sum(1 for _ in handle)
        print(f"{count} {arg}")

    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

import pathlib
import subprocess
import sys

import count_lines


def test_count_lines_multiple_files(tmp_path: pathlib.Path, capsys) -> None:
    first = tmp_path / "first.txt"
    first.write_text("alpha\nbeta\ngamma\n", encoding="utf-8")
    second = tmp_path / "second.txt"
    second.write_text("one\n", encoding="utf-8")
    empty = tmp_path / "empty.txt"
    empty.write_text("", encoding="utf-8")

    exit_code = count_lines.main([str(first), str(second), str(empty)])

    captured = capsys.readouterr()
    assert exit_code == 0
    assert captured.out == f"3 {first}\n1 {second}\n0 {empty}\n"
    assert captured.err == ""


def test_count_lines_missing_file(tmp_path: pathlib.Path, capsys) -> None:
    missing = tmp_path / "missing.txt"

    exit_code = count_lines.main([str(missing)])

    captured = capsys.readouterr()
    assert exit_code == 1
    assert captured.out == ""
    assert f"Error: {missing} is not a file" in captured.err


def test_count_lines_no_arguments(capsys) -> None:
    exit_code = count_lines.main([])

    captured = capsys.readouterr()
    assert exit_code == 1
    assert captured.out == ""
    assert "Usage:" in captured.err


def test_count_lines_cli(tmp_path: pathlib.Path) -> None:
    sample = tmp_path / "sample.txt"
    sample.write_text("line 1\nline 2\n", encoding="utf-8")

    result = subprocess.run(
        [sys.executable, "count_lines.py", str(sample)],
        capture_output=True,
        text=True,
        check=False,
    )

    assert result.returncode == 0
    assert result.stdout == f"2 {sample}\n"
    assert result.stderr == ""
