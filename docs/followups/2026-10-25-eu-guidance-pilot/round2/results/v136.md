from pathlib import Path
import tomllib


def read_config(path: str | Path) -> tuple[int, int]:
    """Read timeout and retries values from a TOML configuration file."""
    config_path = Path(path)
    if not config_path.is_file():
        raise FileNotFoundError(f"Configuration file does not exist: {config_path}")

    with config_path.open("rb") as file:
        data = tomllib.load(file)

    if "timeout" not in data or "retries" not in data:
        raise KeyError("Configuration must include both 'timeout' and 'retries'.")

    timeout = data["timeout"]
    retries = data["retries"]

    if (
        not isinstance(timeout, int)
        or isinstance(timeout, bool)
        or not isinstance(retries, int)
        or isinstance(retries, bool)
    ):
        raise TypeError("Both 'timeout' and 'retries' must be integers.")

    return timeout, retries

from pathlib import Path
import pytest

from config import read_config


def test_read_config_returns_timeout_and_retries(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")

    result = read_config(config_file)

    assert result == (30, 3)


def test_read_config_missing_key_raises_key_error(tmp_path: Path) -> None:
