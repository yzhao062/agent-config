"""Print the number of lines in each file named on the command line."""

import sys


def count_lines(path):
    """Return the number of lines in the file at path, counting bytes so encoding does not matter."""
    count = 0
    last = b""
    with open(path, "rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            count += chunk.count(b"\n")
            last = chunk[-1:]
    if last and last != b"\n":
        count += 1  # final line without a trailing newline
    return count


def main(argv):
    status = 0
    for name in argv:
        try:
            print(f"{count_lines(name)} {name}")
        except OSError as error:
            print(f"count_lines: {name}: {error.strerror}", file=sys.stderr)
            status = 1
    return status


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

from count_lines import count_lines, main


def test_trailing_newline(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"one\ntwo\n")
    assert count_lines(str(f)) == 2


def test_no_trailing_newline(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"one\ntwo")
    assert count_lines(str(f)) == 2


def test_empty_file(tmp_path):
    f = tmp_path / "a.txt"
    f.write_bytes(b"")
    assert count_lines(str(f)) == 0


def test_main_output_and_missing_file(tmp_path, capsys):
    f = tmp_path / "a.txt"
    f.write_bytes(b"x\n")
    missing = str(tmp_path / "nope")
    status = main([str(f), missing])
    out = capsys.readouterr()
    assert out.out == f"1 {f}\n"
    assert missing in out.err
