**Goal:** Ship aa v0.5.0 — direct-URL pack consumption + 4-method auth chain + 3 v0.4.x deferral wirings + `prompt`-default update policy with session-bootstrap banner integration. Acceptance test: `yzhao062/agent-pack@v0.1.0` one-line install end-to-end on Windows + Spark Ubuntu, both Claude Code and Codex drivers.

**Architecture:** Layered functional decomposition adding to the v0.4.0 module split. PyPI vendoring keeps the installed CLI self-contained.

**Tech Stack:** Python 3.9+, PyYAML, `git` subprocess, `gh` CLI, GitHub Actions, pytest, GitHub API.

```python
class AuthChainExhaustedError(Exception):
    """Raised when all auth methods in the chain failed.

    Carries per-method outcomes so the caller can render a composite
    error with each method's status + reason.
    """
    def __init__(self, url: str, ref: str, attempts: list[tuple[str, str]]):
        self.url = url
        self.ref = ref
        self.attempts = attempts  # [(method, "skipped: ..." | "exit N: ...")]
        super().__init__(self._format())

    def _format(self) -> str:
        lines = [f"Failed to resolve {self.url}@{self.ref}"]
        for method, status in self.attempts:
            lines.append(f"  {method:11s} : {status}")
        return "\n".join(lines)


_AUTH_CHAIN_ORDER = ("ssh", "gh", "github_token", "anonymous")


def _git_ls_remote(url: str, ref: str, method: str) -> tuple[bool, str, str]:
    """Run `git ls-remote <url> <ref>` with method-specific config.

    Returns (success, sha_or_empty, error_or_empty).
    """
    args = ["git", "ls-remote"]
    env = noninteractive_fetch_env()

    if method == "ssh":
        # SSH: rewrite URL to git@github.com:owner/repo form for github.com
        owner_repo = canonical_github_identity(url)
        target = f"git@github.com:{owner_repo}.git" if owner_repo else url
    elif method == "gh":
        args = ["git", "-c", "credential.helper=!gh auth git-credential", "ls-remote"]
        target = url if url.startswith("http") else _to_https_url(url)
    elif method == "github_token":
        # Token via inline credential helper that reads from env (no argv leak).
        # ls-remote uses the same helper string approach as clone, since
        # ls-remote also goes through git's credential machinery.
        args = ["git", "-c", f"credential.helper={_token_credential_helper()}", "ls-remote"]
        # The credential helper needs GITHUB_TOKEN in env; ensure it's there
        env["GITHUB_TOKEN"] = os.environ.get("GITHUB_TOKEN", "")
        target = url if url.startswith("http") else _to_https_url(url)
    else:  # anonymous
        target = url if url.startswith("http") else _to_https_url(url)

    try:
        result = subprocess.run(
            args + [target, ref],
            capture_output=True, text=True, timeout=15, env=env,
        )
    except (FileNotFoundError, subprocess.TimeoutExpired) as e:
        return False, "", str(e)

    if result.returncode != 0:
        return False, "", redact_secret_text(result.stderr.strip())

    # ls-remote output: "<sha>\trefs/heads/main\n" or similar
    line = result.stdout.strip().split("\n")[0]
    parts = line.split("\t")
    if len(parts) != 2 or len(parts[0]) != 40:
        return False, "", f"unexpected ls-remote output: {line!r}"
    return True, parts[0], ""


def _to_https_url(url: str) -> str:
    """Convert any github.com URL form to https://github.com/owner/repo."""
    owner_repo = canonical_github_identity(url)
    if owner_repo:
        return f"https://github.com/{owner_repo}"
    return url


def resolve_ref_with_auth_chain(
    url: str, ref: str, *, explicit_method: str | None = None
) -> tuple[str, str]:
    """Resolve `(url, ref)` to a 40-char commit sha via the auth chain.

    Returns (resolved_commit, method_that_succeeded).
    Raises AuthChainExhaustedError if no method succeeds (or if explicit
    method fails).
    """
    methods = (explicit_method,) if explicit_method else _AUTH_CHAIN_ORDER
    attempts: list[tuple[str, str]] = []

    for method in methods:
        # Probe (fast skip)
        if method == "ssh" and not ssh_agent_available():
            attempts.append((method, "skipped (ssh-add -l reported no identities)"))
            continue
        if method == "gh" and not gh_cli_authenticated():
            attempts.append((method, "skipped (gh auth status returned non-zero)"))
            continue
        if method == "github_token" and not github_token_available():
            attempts.append((method, "skipped (GITHUB_TOKEN env var not set)"))
            continue

        success, sha, err = _git_ls_remote(url, ref, method)
        if success:
            return sha, method
        attempts.append((method, f"attempted: {err or 'unknown failure'}"))

    raise AuthChainExhaustedError(url, ref, attempts)
```

```python
import tempfile
import pathlib
import shutil
import stat


def _git_rev_parse_head(repo_dir: pathlib.Path) -> str:
    """Return the 40-char HEAD commit of a clone."""
    result = subprocess.run(
        ["git", "rev-parse", "HEAD"],
        cwd=repo_dir, capture_output=True, text=True, check=True,
    )
    return result.stdout.strip()


def _write_git_askpass_helper(token_env_var: str = "GITHUB_TOKEN") -> pathlib.Path:
    """Write a temporary GIT_ASKPASS helper script reading token from env.

    The helper outputs `x-access-token` for username prompt and
    `$GITHUB_TOKEN` for password prompt. Token is NEVER passed via argv.
    """
    tmp = pathlib.Path(tempfile.mkdtemp(prefix="aa-askpass-"))
    if os.name == "nt":
        # Windows: use a .bat that reads from env. %TOKEN_ENV_VAR% expands
        # at runtime to the actual env value (not the literal name).
        helper = tmp / "askpass.bat"
        helper.write_text(
            "@echo off\r\n"
            "echo %~1 | findstr /C:\"Username\" >nul && (echo x-access-token) "
            "|| (echo %{token_env_var}%)\r\n".format(token_env_var=token_env_var)
        )
    else:
        # POSIX: shell variable expansion happens at script-run time.
        # The Python f-string injects only the var NAME into the script
        # body; the expansion to its value happens when git invokes the
        # script, NOT when Python writes the script.
        helper = tmp / "askpass.sh"
        helper.write_text(
            "#!/bin/sh\n"
            "case \"$1\" in\n"
            "  *Username*) printf '%s\\n' x-access-token ;;\n"
            f"  *) printf '%s\\n' \"${{{token_env_var}}}\" ;;\n"
            "esac\n",
            encoding="utf-8",
        )
        helper.chmod(0o700)
    return helper


def _token_credential_helper() -> str:
    """Return a git-credential-helper inline script that reads GITHUB_TOKEN.

    Used by ls-remote (Task 2.2) where a credential helper string is
    needed without writing a separate file. The returned string is
    embedded in `git -c credential.helper=<this>`.
    """
    return "!f() { echo username=x-access-token; echo password=$GITHUB_TOKEN; }; f"


def fetch_with_method(
    url: str, ref: str, method: str, *,
    dest: pathlib.Path | None = None,
) -> "PackArchive":  # forward-ref to source_fetch.PackArchive (Phase 3)
    """Run a `git clone` subprocess for one auth method.

    Returns a PackArchive (resolved_commit, archive_dir, method, ...).
    Raises subprocess.CalledProcessError on failure (caller wraps).
    """
    from scripts.packs import source_fetch  # forward import to avoid cycles

    if dest is None:
        dest = pathlib.Path(tempfile.mkdtemp(prefix="aa-clone-"))

    base_args = ["git"]
    env = noninteractive_fetch_env()
    askpass_helper: pathlib.Path | None = None

    if method == "ssh":
        owner_repo = canonical_github_identity(url)
        target = f"git@github.com:{owner_repo}.git" if owner_repo else url
    elif method == "gh":
        base_args = ["git", "-c", "credential.helper=!gh auth git-credential"]
        target = _to_https_url(url)
    elif method == "github_token":
        askpass_helper = _write_git_askpass_helper()
        env["GIT_ASKPASS"] = str(askpass_helper)
        env["GIT_TERMINAL_PROMPT"] = "0"
        target = _to_https_url(url)
    elif method == "anonymous":
        target = _to_https_url(url) if url.startswith("git@") else url
    else:
        raise ValueError(f"unknown auth method: {method!r}")

    clone_args = base_args + [
        "clone", "--depth=1", "--filter=blob:none",
        "--branch", ref, target, str(dest),
    ]

    try:
        subprocess.run(
            clone_args, env=env, check=True, capture_output=True, text=True,
            timeout=120,
        )
        resolved_commit = _git_rev_parse_head(dest)
    finally:
        if askpass_helper is not None:
            shutil.rmtree(askpass_helper.parent, ignore_errors=True)

    return source_fetch.PackArchive(
        url=url, ref=ref, resolved_commit=resolved_commit,
        method=method, archive_dir=dest,
        canonical_id=canonical_github_identity(url),
        cache_key=source_fetch.compute_cache_key(url, resolved_commit),
    )
```

```python
def fetch_with_auth_chain(
    url: str, ref: str, *,
    explicit_method: str | None = None,
    cache_root: pathlib.Path | None = None,
) -> "source_fetch.PackArchive":
    """Walk the auth chain; return the first PackArchive that succeeds.

    Order (when explicit_method=None): ssh → gh → github_token → anonymous.
    Methods whose probe fails are skipped (logged in the AuthChainExhausted
    composite). With explicit_method set, the chain is bypassed entirely:
    only that method is attempted; failure raises (no anonymous fallback).
    """
    from scripts.packs import source_fetch  # forward import

    methods = (explicit_method,) if explicit_method else _AUTH_CHAIN_ORDER
    attempts: list[tuple[str, str]] = []

    for method in methods:
        if method == "ssh" and not ssh_agent_available():
            attempts.append((method, "skipped (ssh-add -l reported no identities)"))
            continue
        if method == "gh" and not gh_cli_authenticated():
            attempts.append((method, "skipped (gh auth status returned non-zero)"))
            continue
        if method == "github_token" and not github_token_available():
            attempts.append((method, "skipped (GITHUB_TOKEN env var not set)"))
            continue

        try:
            archive = fetch_with_method(url, ref, method)
            return archive
        except subprocess.CalledProcessError as e:
            stderr = redact_secret_text((e.stderr or "").strip())
            attempts.append((method, f"attempted, exit {e.returncode}: {stderr}"))
            continue

    raise AuthChainExhaustedError(url, ref, attempts)
```

```python
# ======================================================================
# Secret redaction (v0.5.0 H2)
# ======================================================================

import re

_TOKEN_PREFIX_PATTERN = re.compile(
    r"\b(?:ghp_|gho_|ghu_|ghr_|ghs_|github_pat_)[A-Za-z0-9_]+",
)
_BEARER_PATTERN = re.compile(r"Bearer\s+[A-Za-z0-9._\-]+")


def redact_url_userinfo(url: str) -> str:
    """Redact userinfo (user / password / token) in a URL.

    HTTPS forms with any userinfo → 'https://<redacted>@host/path'.
    SSH transport-only ('git@host:...') is unchanged.
    SSH with password ('ssh://user:pass@host/...') redacts the password.
    """
    if not isinstance(url, str) or not url:
        return url

    # scp-style SSH (git@host:path) — transport username only
    if url.startswith("git@") and "://" not in url:
        return url

    # https://...@host/...
    https_match = re.match(r"^(https?)://([^/@]+)@(.+)$", url)
    if https_match:
        scheme, _userinfo, rest = https_match.groups()
        return f"{scheme}://<redacted>@{rest}"

    # ssh://user:pass@host/... or git+ssh://user:pass@host/...
    ssh_match = re.match(r"^(ssh|git\+ssh)://([^:@]+):([^@]+)@(.+)$", url)
    if ssh_match:
        scheme, user, _pass, rest = ssh_match.groups()
        return f"{scheme}://{user}:<redacted>@{rest}"

    # ssh://user@host/... (no password) — leave unchanged
    return url


def redact_secret_text(
    text: str,
    *,
    known_secrets: list[str] | tuple[str, ...] = (),
) -> str:
    """Redact known secrets and recognized GitHub token patterns from text.

    Used to sanitize subprocess stderr/stdout, exception messages, and
    state-file content before display or persistence.
    """
    if not isinstance(text, str):
        return text
    out = text
    for secret in known_secrets:
        if secret:
            out = out.replace(secret, "<redacted>")
    out = _TOKEN_PREFIX_PATTERN.sub("<redacted>", out)
    out = _BEARER_PATTERN.sub("Bearer <redacted>", out)
    return out
```

```python
"""Source fetch v0.5.0 — fetch a pack archive from a URL with auth chain."""
from __future__ import annotations

import hashlib
import pathlib
import shutil
from dataclasses import dataclass
from typing import Literal

from scripts.packs import auth


@dataclass(frozen=True)
class PackArchive:
    url: str
    ref: str
    resolved_commit: str
    method: str
    archive_dir: pathlib.Path
    canonical_id: str | None
    cache_key: str


class PackLockDriftError(Exception):
    """Raised when update_policy=locked detects ref drift vs pack-lock."""

    def __init__(self, url: str, ref: str, recorded: str, current: str):
        self.url = url
        self.ref = ref
        self.recorded = recorded
        self.current = current
        super().__init__(
            f"pack-lock drift for {url}@{ref}: recorded {recorded}, "
            f"current upstream {current}. Run `anywhere-agents pack update "
            f"<name>` to accept the new commit, or pin ref to {recorded}."
        )


def compute_cache_key(url: str, resolved_commit: str) -> str:
    canonical = auth.canonical_github_identity(url)
    if canonical:
        prefix = hashlib.sha256(canonical.encode()).hexdigest()[:16]
    else:
        prefix = hashlib.sha256(url.encode()).hexdigest()[:16]
    return f"{prefix}/{resolved_commit}"


def _iter_content_files(archive_dir: pathlib.Path):
    """Yield only content files (skip .git/ metadata and .dir-sha256 marker).

    Without this filter, the dir-sha256 marker file would be included in
    its own hash on subsequent runs → cache-hit integrity check would
    always fail. .git/ is clone metadata, not pack content.
    """
    for path in archive_dir.rglob("*"):
        if not path.is_file():
            continue
        rel = path.relative_to(archive_dir)
        if rel.parts and rel.parts[0] == ".git":
            continue
        if rel.as_posix() == ".dir-sha256":
            continue
        yield path


def _compute_dir_sha256(archive_dir: pathlib.Path) -> str:
    """Stable Merkle hash over the archive directory's content files."""
    paths = sorted(_iter_content_files(archive_dir))
    h = hashlib.sha256()
    for p in paths:
        rel = p.relative_to(archive_dir).as_posix()
        h.update(rel.encode("utf-8"))
        h.update(b"\0")
        h.update(p.read_bytes())
        h.update(b"\0")
    return f"dir-sha256:{h.hexdigest()}"


def fetch_pack(
    url: str,
    ref: str,
    *,
    policy: Literal["auto", "prompt", "locked"] = "prompt",
    explicit_auth: str | None = None,
    pack_lock_recorded_commit: str | None = None,
    cache_root: pathlib.Path | None = None,
) -> PackArchive:
    """Fetch (or cache-hit) the pack archive at url@ref.

    Ordering (Codex Round 1 M5):
      1. Validate URL via auth.reject_credential_url (no network).
      2. Resolve ref to commit sha via auth.resolve_ref_with_auth_chain.
      3. Compare against pack_lock_recorded_commit; under locked drift,
         raise PackLockDriftError.
      4. Cache lookup at (canonical_id_or_url_hash, resolved_commit).
         Hit → re-verify dir-sha256, return.
      5. Cache miss → auth.fetch_with_auth_chain into staging; rename
         atomically into cache; return.
    """
    if cache_root is None:
        cache_root = pathlib.Path(".agent-config/cache")
    cache_root.mkdir(parents=True, exist_ok=True)

    # 1. Validate
    auth.reject_credential_url(url, source_layer="source_fetch")

    # 2. Resolve
    resolved_commit, _resolve_method = auth.resolve_ref_with_auth_chain(
        url, ref, explicit_method=explicit_auth)

    # 3. Compare against pack-lock
    if (
        pack_lock_recorded_commit
        and resolved_commit != pack_lock_recorded_commit
    ):
        if policy == "locked":
            raise PackLockDriftError(
                url, ref, pack_lock_recorded_commit, resolved_commit)
        # policy in {"auto", "prompt"}: drift handled by compose layer

    # 4. Cache lookup
    cache_key = compute_cache_key(url, resolved_commit)
    cache_dir = cache_root / cache_key
    if cache_dir.exists():
        # Re-verify integrity — drop and refetch on mismatch
        recorded_sha = (cache_dir / ".dir-sha256").read_text() if (cache_dir / ".dir-sha256").exists() else None
        current_sha = _compute_dir_sha256(cache_dir)
        if recorded_sha == current_sha:
            return PackArchive(
                url=url, ref=ref, resolved_commit=resolved_commit,
                method="cached", archive_dir=cache_dir,
                canonical_id=auth.canonical_github_identity(url),
                cache_key=cache_key,
            )
        # Integrity mismatch — clear and re-fetch (one retry)
        shutil.rmtree(cache_dir, ignore_errors=True)

    # 5. Cache miss — fetch
    archive = auth.fetch_with_auth_chain(url, ref, explicit_method=explicit_auth)
    # Move staging into cache atomically
    cache_dir.parent.mkdir(parents=True, exist_ok=True)
    if cache_dir.exists():
        shutil.rmtree(cache_dir, ignore_errors=True)
    shutil.move(str(archive.archive_dir), str(cache_dir))
    # Record dir-sha256 for next-run integrity check
    sha = _compute_dir_sha256(cache_dir)
    (cache_dir / ".dir-sha256").write_text(sha)

    return PackArchive(
```

```python
def _resolve_passive_body(
    mapping: dict,
    pack_manifest: dict,
    ctx,
    *,
    cache_dir: pathlib.Path,
    no_cache: bool,
) -> tuple[str, str]:
    """Resolve one passive file mapping from archive or v0.4 raw URL path.

    When ctx.pack_source_dir is set (v0.5.0 inline-source path), read
    mapping['from'] from that directory and hash the bytes — no network.
    Otherwise fall back to v0.4.0 legacy raw-URL fetch (preserves
    bundled agent-style behavior).
    """
    from_path = mapping["from"]

    if getattr(ctx, "pack_source_dir", None) is not None:
        # v0.5.0 archive-backed read
        body = (ctx.pack_source_dir / from_path).read_text(encoding="utf-8")
        sha = hashlib.sha256(body.encode("utf-8")).hexdigest()
        return body, sha

    # v0.4.0 legacy raw-URL path (bundled agent-style etc.)
    pack_name = pack_manifest["name"]
    source = pack_manifest.get("source")
    repo_url = source.get("repo") if isinstance(source, dict) else source
    ref = source.get("ref") if isinstance(source, dict) else pack_manifest.get("default-ref")
    if not isinstance(ref, str) or not ref:
        raise ValueError(
            f"pack {pack_name!r} passive entry: 'source.ref' required for legacy fetch")
    fetch_url = _derive_raw_url(repo_url, ref, from_path)
    _legacy.validate_ref(pack_name, ref)

    content, sha = _legacy.fetch_rule_pack(fetch_url, ref, cache_md, no_cache)
    _legacy.validate_rule_pack(pack_name, content)
```

```python
def reconcile_orphans(
    project_root: pathlib.Path,
    user_root: pathlib.Path,
    *,
    locks_held: bool = False,
    drift_callback: Callable[[OrphanClassification], None] | None = None,
) -> ReconciliationReport:
    if not locks_held:
        with (
            locks.acquire(locks.user_lock_path(user_root), timeout=30),
            locks.acquire(locks.repo_lock_path(project_root), timeout=30),
        ):
            return _reconcile_inner(project_root, user_root, drift_callback)
    # locks_held=True: caller (compose_packs) holds the locks. The
    # classifier needs to know self vs foreign — handled by passing
    # the current process's lock-context into scan_orphans (added in
    # a sibling task that extends classify_orphan(), since shipped
    # OrphanClassification has no holder_pid field). For v0.5.0,
    # rely on the classifier extension rather than a post-hoc PID check.
    return _reconcile_inner(project_root, user_root, drift_callback)


def _reconcile_inner(project_root, user_root, drift_callback):
    staging_dirs = _collect_staging_dirs(project_root, user_root)
    report = ReconciliationReport()
    for orphan in scan_orphans(staging_dirs):
        if orphan.label == LIVE:
            report.live.append(orphan.staging_dir)
        elif orphan.label == ROLLBACK_OK:
            cleanup_staging(orphan.staging_dir)
            report.rolled_back.append(orphan.staging_dir)
        elif orphan.label == ROLLFORWARD_OK:
            cleanup_staging(orphan.staging_dir)
            report.rolled_forward.append(orphan.staging_dir)
        elif orphan.label == PARTIAL:
            if _can_reapply_partial(orphan):
                _reapply_partial(orphan)
                cleanup_staging(orphan.staging_dir)
                report.partial_reapplied.append(orphan.staging_dir)
            else:
                if drift_callback:
                    drift_callback(orphan)
                report.blocking.append(orphan.staging_dir)
        elif orphan.label in (DRIFT, MALFORMED):
            if drift_callback:
                drift_callback(orphan)
            report.blocking.append(orphan.staging_dir)
    return report
```

```python
#!/usr/bin/env python3
"""Vendor scripts/packs/{auth,source_fetch,schema}.py into the PyPI package.

Source of truth: scripts/packs/*.py
Vendored copies: packages/pypi/anywhere_agents/packs/*.py

CI runs this and fails if the vendored copies don't match.
"""
import pathlib, shutil, sys

REPO = pathlib.Path(__file__).resolve().parent.parent
SRC = REPO / "scripts" / "packs"
DST = REPO / "packages" / "pypi" / "anywhere_agents" / "packs"
MODULES = ("auth.py", "source_fetch.py", "schema.py")


def vendor():
    DST.mkdir(parents=True, exist_ok=True)
    (DST / "__init__.py").write_text(
        '"""Vendored copies of scripts/packs/* for the installed CLI."""\n')
    for name in MODULES:
        src_file = SRC / name
        dst_file = DST / name
        # Rewrite imports: scripts.packs.X → anywhere_agents.packs.X
        body = src_file.read_text(encoding="utf-8")
        body = body.replace(
            "from scripts.packs",
            "from anywhere_agents.packs",
        )
        body = body.replace(
            "import scripts.packs",
            "import anywhere_agents.packs",
        )
        dst_file.write_text(body, encoding="utf-8")


def _vendored_text(src_file: pathlib.Path) -> str:
    body = src_file.read_text(encoding="utf-8")
    body = body.replace("from scripts.packs", "from anywhere_agents.packs")
    body = body.replace("import scripts.packs", "import anywhere_agents.packs")
    return body


def check() -> None:
    """Fail with non-zero exit if vendored copies don't match the source.

    Compares the destination files byte-for-byte against the rewrite
    that would be produced by `vendor()`. Used by CI vendor-sync job.
    """
    errors = []
    expected_init = '"""Vendored copies of scripts/packs/* for the installed CLI."""\n'
    init_path = DST / "__init__.py"
    if not init_path.exists() or init_path.read_text(encoding="utf-8") != expected_init:
        errors.append(str(init_path))
    for name in MODULES:
        expected = _vendored_text(SRC / name)
        actual_path = DST / name
        actual = actual_path.read_text(encoding="utf-8") if actual_path.exists() else ""
        if actual != expected:
            errors.append(str(actual_path))
    if errors:
        print("vendored pack modules are out of sync with scripts/packs/*:", file=sys.stderr)
        for path in errors:
            print(f"  {path}", file=sys.stderr)
        print("Run `python scripts/vendor-packs.py` to regenerate.", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    if sys.argv[1:2] == ["check"]:
        check()
    else:
        vendor()
```

```python
    """Update a pack's user-config ref pin and trigger project compose."""
    from anywhere_agents.packs import auth, source_fetch
```
