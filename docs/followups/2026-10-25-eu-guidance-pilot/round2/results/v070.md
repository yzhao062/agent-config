#!/usr/bin/env python3
"""Count lines in files named on the command line."""

import sys


def count_lines(filepath: str) -> int:
    """Return the number of lines in the specified file."""
    with open(filepath, "r", encoding="utf-8", errors="replace") as file_handle:
        return sum(1 for _ in file_handle)


def main(args: list[str] | None = None) -> int:
    """Process command-line arguments and print line counts."""
    if args is None:
        args = sys.argv[1:]

    exit_code = 0
    for filename in args:
        try:
            line_count = count_lines(filename)
            print(f"{line_count} {filename}")
        except OSError as error:
            sys.stderr.write(f"count_lines: {filename}: {error.strerror or error}\n")
            exit_code = 1

    return exit_code


if __name__ == "__main__":
    sys.exit(main())
