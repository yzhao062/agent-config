FILE: hook.py
```python
import json
import sys


def main() -> None:
    try:
        payload = json.load(sys.stdin)
    except json.JSONDecodeError:
        payload = {}

    command = payload.get("command", "")
    decision = "deny" if "rm -rf /" in command else "allow"
    print(json.dumps({"decision": decision}))


if __name__ == "__main__":
    main()
```

```python
import json
import subprocess
import sys


def run_hook(stdin_text: str) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        [sys.executable, "hook.py"],
        input=stdin_text,
        text=True,
        capture_output=True,
        check=False,
    )


def test_allows_safe_command() -> None:
    result = run_hook('{"tool": "Bash", "command": "ls -la"}')
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "allow"}


def test_denies_rm_rf_root() -> None:
    result = run_hook('{"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"}')
    assert result.returncode == 0
    assert json.loads(result.stdout) == {"decision": "deny"}


def test_allows_empty_or_truncated_payload() -> None:
    for payload in ["", '{"tool": "Bash", "command": "ls']:
        result = run_hook(payload)
        assert result.returncode == 0
        assert json.loads(result.stdout) == {"decision": "allow"}
```
