import sys


def main() -> None:
    for filename in sys.argv[1:]:
        with open(filename, "r", encoding="utf-8") as file:
            line_count = sum(1 for _ in file)
        print(f"{line_count} {filename}")


if __name__ == "__main__":
    main()
