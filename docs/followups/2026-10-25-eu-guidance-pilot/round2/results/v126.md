```python
from pathlib import Path
import tomllib


def read_config(path: str | Path) -> tuple[int, int]:
    """Read a TOML configuration file and return (timeout, retries) as integers."""
    file_path = Path(path)
    with file_path.open("rb") as f:
        data = tomllib.load(f)

    timeout = data["timeout"]
    retries = data["retries"]

    if isinstance(timeout, bool) or not isinstance(timeout, int):
        raise ValueError("timeout must be an integer")
    if isinstance(retries, bool) or not isinstance(retries, int):
        raise ValueError("retries must be an integer")

    return (timeout, retries)
```

```python
from pathlib import Path
import tomllib
import pytest

from config import read_config


def test_read_config_valid(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 30\nretries = 3\n", encoding="utf-8")

    result = read_config(config_file)
    assert result == (30, 3)
    assert isinstance(result, tuple)
    assert isinstance(result[0], int)
    assert isinstance(result[1], int)


def test_read_config_string_path(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 10\nretries = 5\n", encoding="utf-8")

    result = read_config(str(config_file))
    assert result == (10, 5)


def test_read_config_missing_key(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
    config_file.write_text("timeout = 30\n", encoding="utf-8")

    with pytest.raises(KeyError):
        read_config(config_file)


def test_read_config_invalid_type(tmp_path: Path) -> None:
    config_file = tmp_path / "config.toml"
```
