"""Print the number of lines in each file named on the command line."""

import sys


def count_lines(path):
    """Return the number of lines in the file at path."""
    count = 0
    with open(path, "rb") as handle:
        for _ in handle:
            count += 1
    return count


def main(argv=None):
    """Print one '<count> <name>' line per file. Return an exit status."""
    names = sys.argv[1:] if argv is None else argv
    status = 0
    for name in names:
        try:
            print(f"{count_lines(name)} {name}")
        except OSError as error:
            print(f"count_lines: {name}: {error.strerror}", file=sys.stderr)
            status = 1
    return status


if __name__ == "__main__":
    sys.exit(main())

import count_lines


def test_counts_lines(tmp_path, capsys):
    path = tmp_path / "a.txt"
    path.write_text("one\ntwo\nthree\n")
    assert count_lines.main([str(path)]) == 0
    assert capsys.readouterr().out == f"3 {path}\n"


def test_no_trailing_newline(tmp_path):
    path = tmp_path / "b.txt"
    path.write_text("one\ntwo")
    assert count_lines.count_lines(str(path)) == 2


def test_empty_file(tmp_path):
    path = tmp_path / "c.txt"
    path.write_text("")
    assert count_lines.count_lines(str(path)) == 0


def test_multiple_files_in_order(tmp_path, capsys):
    first = tmp_path / "x"
    second = tmp_path / "y"
    first.write_text("a\n")
    second.write_text("a\nb\n")
    count_lines.main([str(first), str(second)])
    assert capsys.readouterr().out == f"1 {first}\n2 {second}\n"


def test_missing_file(tmp_path, capsys):
    missing = tmp_path / "missing"
    good = tmp_path / "good"
    good.write_text("a\n")
    assert count_lines.main([str(missing), str(good)]) == 1
    captured = capsys.readouterr()
    assert captured.out == f"1 {good}\n"
    assert "missing" in captured.err
