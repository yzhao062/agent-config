import json
import sys


def decide(payload_text: str) -> str:
    """Determine whether to allow or deny a tool call."""
    if not payload_text.strip():
        return "allow"

    try:
        data = json.loads(payload_text)
    except Exception:
        # When JSON is malformed or truncated, check if the raw text contains the forbidden command.
        if "rm -rf /" in payload_text:
            return "deny"
        return "allow"

    if isinstance(data, dict):
        command = data.get("command")
        if isinstance(command, str):
            if "rm -rf /" in command:
                return "deny"
            return "allow"
        if isinstance(command, list):
            joined_command = " ".join(str(item) for item in command)
            if "rm -rf /" in joined_command:
                return "deny"
            return "allow"
        return "allow"

    if "rm -rf /" in payload_text:
        return "deny"

    return "allow"


def main() -> None:
    """Read the payload from standard input and print the decision."""
    try:
        payload_text = sys.stdin.read()
        decision = decide(payload_text)
    except Exception:
        decision = "allow"

    print(json.dumps({"decision": decision}))


if __name__ == "__main__":
    main()

import json
import subprocess
import sys
from pathlib import Path

import pytest
from hook import decide

HOOK_SCRIPT = Path(__file__).parent / "hook.py"


def run_hook_process(input_data: str) -> subprocess.CompletedProcess[str]:
    """Execute hook.py as a subprocess with the given input string."""
    return subprocess.run(
        [sys.executable, str(HOOK_SCRIPT)],
        input=input_data,
        capture_output=True,
        text=True,
        check=False,
    )


@pytest.mark.parametrize(
    ("payload", "expected_decision"),
    [
        ('{"tool": "Bash", "command": "ls -la"}', "allow"),
        ('{"tool": "Bash", "command": "rm -rf /"}', "deny"),
        ('{"tool": "Bash", "command": "sudo rm -rf / --no-preserve-root"}', "deny"),
        ('{"tool": "Bash", "command": "echo safe", "note": "rm -rf /"}', "allow"),
        ("{}", "allow"),
        ('{"tool": "Bash"}', "allow"),
        ('{"tool": "Bash", "command": null}', "allow"),
        ("", "allow"),
        ("   \n\t ", "allow"),
        ('{"tool": "Bash", "command": "ls', "allow"),
        ('{"tool": "Bash", "command": "rm -rf /', "deny"),
        ("not json", "allow"),
        ("not json with rm -rf / included", "deny"),
    ],
)
def test_decide(payload: str, expected_decision: str) -> None:
    assert decide(payload) == expected_decision


def test_command_as_list() -> None:
    payload = json.dumps({"command": ["sudo", "rm -rf /"]})
    assert decide(payload) == "deny"

    safe_payload = json.dumps({"command": ["ls", "-la"]})
    assert decide(safe_payload) == "allow"


@pytest.mark.parametrize(
    ("payload", "expected_decision"),
    [
        ('{"tool": "Bash", "command": "ls -la"}', "allow"),
        ('{"tool": "Bash", "command": "rm -rf /"}', "deny"),
        ("", "allow"),
        ("   ", "allow"),
        ('{"tool": "Bash", "command": "rm -rf /', "deny"),
        ('{"tool": "Bash", "command": "echo', "allow"),
    ],
)
def test_subprocess_execution(payload: str, expected_decision: str) -> None:
    result = run_hook_process(payload)
    assert result.returncode == 0
    assert result.stderr == ""

    output_json = json.loads(result.stdout)
    assert output_json == {"decision": expected_decision}
